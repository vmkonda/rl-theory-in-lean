\title{Rates of Convergence of Actor-Critic and Actor-Only Methods}
\author{Vijaymohan Konda}
\date{November 2026}

\begin{document}
\maketitle

\begin{abstract}
We compare the asymptotic rates of convergence of actor-only and actor-critic methods on
finite discounted Markov decision processes, as a function of the critic's features and of
$\lambda$ in TD($\lambda$). The iterations are linearized around their limit by a first-order
Taylor expansion. This gives linear two-time-scale iterations with Markov noise, whose
asymptotic covariance is given by the two-time-scale note (including the multiplicative noise
that the linearization produces). The figure of merit is the loss in expected total reward,
$\E[J(\theta^*) - J(\theta_k)]$, which to second order is the variance of the parameter weighted
by the Hessian of $J$. We prove that a TD(1) critic, whatever its features, gives exactly the rate
of actor-only with Monte Carlo returns, and that a critic with only the compatible features gives
exactly the rate of actor-only with the same trace parameter. When the critic's state features can represent the value function
(for instance tabular), the method is unbiased for every $\lambda$; in our instances its loss is then
0.2--0.6 of the actor-only loss, decreasing with $\lambda$.
\end{abstract}

\section{Setting}
%======================================================================

\paragraph{The MDP and the policies.} A finite MDP with states $s$, actions $a$, transition
probabilities $P(s'\mid s,a)$, rewards $g(s,a)$, initial law $p_0$ and discount $\gamma \in (0,1)$.
A family of policies $\pi_\theta(a\mid s)$, $\theta \in \R^n$, smooth in $\theta$, with score
$\psi_\theta(s,a) = \nabla_\theta\log\pi_\theta(a\mid s)$; note that
$\sum_a\pi_\theta(a\mid s)\psi_\theta(s,a) = 0$ for every $s$. The objective is the expected
discounted reward $J(\theta) = \E\big[\sum_t\gamma^tg(s_t,a_t)\big]$ with $s_0 \sim p_0$.

\begin{assumption}[Well-posedness]\label{ass:wp}
$J$ has a strict local maximum $\theta^*$ in the interior, with
$H_J := -\nabla^2J(\theta^*) \succ 0$.
\end{assumption}

\paragraph{Sampling.} Both methods sample the state-action chain with restarts: from $(s,a)$,
with probability $\gamma$ the next pair follows the MDP and the policy, and with probability
$1 - \gamma$ the process restarts from $p_0$. Let $Y_k$ be the pair at time $k$ together with a flag
recording a restart, and $c_{k+1} = 1$ if $Y_{k+1}$ is not a restart, $0$ otherwise. The pieces
between restarts are \emph{episodes}. The stationary law $\nu_\theta$ of $Y$ is the discounted
occupancy measure, so $\E_\nu[\psi\,G] = (1-\gamma)\nabla J(\theta)$, where $G_k$ is the
reward-to-go of the episode, the sum of $g(Y_j)$ over the rest of the episode from $k$ on.

\paragraph{Actor-only.} With the trace $Z_k = \lambda c_kZ_{k-1} + \psi(Y_k)$,
\[
  \theta_{k+1} = \theta_k + \beta_k\,g(Y_k)Z_k .
\]
For $\lambda = 1$ this is REINFORCE in per-step form: summed over an episode, $\sum_kg(Y_k)Z_k
= \sum_k\psi(Y_k)G_k$.

\paragraph{Actor-critic.} The critic has features $\phi_\theta(s,a) = (\psi_\theta(s,a), \bar\phi(s))$,
the compatible features plus extra state features, a trace $z_k = \lambda c_kz_{k-1} + \phi(Y_k)$
and the TD error $d_k = g(Y_k) + c_{k+1}r_k^\top\phi(Y_{k+1}) - r_k^\top\phi(Y_k)$:
\[
  r_{k+1} = r_k + \gamma_k\,d_kz_k,\qquad \theta_{k+1} = \theta_k + \beta_k\,\big(r_k^\top\phi(Y_k)\big)\psi(Y_k),
\]
with $\beta_k/\gamma_k \to 0$ (Konda 2002, Chapter 5). For fixed $\theta$ the critic converges to the
TD($\lambda$) solution $r^*(\theta)$ of $A_c r + b = 0$, where
$A_c = \E_\nu[z_k(c_{k+1}\phi(Y_{k+1}) - \phi(Y_k))^\top]$ and $b = \E_\nu[z_kg(Y_k)]$, and the actor
follows the mean direction $f(\theta) = C\,r^*(\theta)$ with $C = \E_\nu[\psi\phi^\top]$.

\section{Linearization}
%======================================================================

Write either method as
\[
  \theta_{k+1} = \theta_k + \beta_kF(\theta_k, r_k, \xi_k),\qquad
  r_{k+1} = r_k + \gamma_kG(\theta_k, r_k, \xi_k),
\]
where $\xi_k = (Y_k, Y_{k+1}, z_k)$ and the chain $\xi$ is generated under $\pi_{\theta_k}$ (for
actor-only there is no $r$). Let $\bar F(\theta, r)$ and $\bar G(\theta, r)$ be the means under the
stationary law of the chain at $\theta$, $r^*(\theta)$ the root of $\bar G(\theta, \cdot)$, and
$\hat\theta$ a root of $f(\theta) = \bar F(\theta, r^*(\theta))$ with $\hat r = r^*(\hat\theta)$.

\paragraph{Taylor expansion.} To first order around $(\hat\theta, \hat r)$,
\[
  \bar F(\theta, r) = -A_{11}(\theta - \hat\theta) - A_{12}(r - \hat r) + O(\|\cdot\|^2),\qquad
  \bar G(\theta, r) = -A_{21}(\theta - \hat\theta) - A_{22}(r - \hat r) + O(\|\cdot\|^2),
\]
with $A_{11} = -\partial_\theta\bar F$, $A_{12} = -\partial_r\bar F$, $A_{21} = -\partial_\theta\bar G$ and
$A_{22} = -\partial_r\bar G$ at $(\hat\theta, \hat r)$. To zeroth order the noise is
$F(\theta, r, \xi) - \bar F(\theta, r) = V(\xi) + O(\|\cdot\|)$ with $V(\xi) = F(\hat\theta, \hat r, \xi)$, and
similarly $W(\xi) = G(\hat\theta, \hat r, \xi)$. Dropping the remainders, $x_k = \theta_k - \hat\theta$ and
$y_k = r_k - \hat r$ follow the linear two-time-scale iteration
\begin{equation}\label{eq:lin}
\begin{aligned}
  x_{k+1} &= x_k - \beta_k(A_{11}x_k + A_{12}y_k) + \beta_kV_k,\\
  y_{k+1} &= y_k - \gamma_k(A_{21}x_k + A_{22}y_k) + \gamma_kW_k,
\end{aligned}
\end{equation}
driven by the chain at $\hat\theta$. For actor-only it is the single-time-scale iteration
$x_{k+1} = x_k - \beta_kHx_k + \beta_kV_k$ with $H = -f'(\hat\theta)$.

The first-order terms of the noise, $(\partial F(\xi) - \partial\bar F)\cdot(x, y)$, are multiplicative
noise. By Theorem~4.1 of the two-time-scale note they do not change the asymptotic covariance,
which is why the zeroth-order noise suffices. For actor-critic, $\partial_rF = \psi\phi^\top$ and
$\partial_rG = z(c'\phi' - \phi)^\top$ are affine in the trace and have the Poisson decompositions of
Section~4 of the single-time-scale note.

\begin{lemma}\label{lem:delta}
$A_{22} = -A_c$, $A_{12} = -C$, and
$\Delta := A_{11} - A_{12}A_{22}^{-1}A_{21} = -f'(\hat\theta) =: H$, the Jacobian of the actor's mean
direction with the critic at its fixed point.
\end{lemma}
\begin{proof}
$\bar G(\theta, r) = A_cr + b$ and $\bar F(\theta, r) = Cr$ at fixed $\theta$, which gives the first two
identities. Differentiating $\bar G(\theta, r^*(\theta)) = 0$ gives
$\partial_\theta r^* = -(\partial_r\bar G)^{-1}\partial_\theta\bar G = -A_{22}^{-1}A_{21}$, so
$f'(\hat\theta) = \partial_\theta\bar F + \partial_r\bar F\,\partial_\theta r^* = -A_{11} + A_{12}A_{22}^{-1}A_{21}$.
\end{proof}

\begin{assumption}[Linearization]\label{ass:lin}
The asymptotic covariance of $\theta_k - \hat\theta$ for the actual iteration equals that of $x_k$ in
\eqref{eq:lin}.
\end{assumption}

This is the standard step from the ODE method to rates (Benveniste, M\'etivier and Priouret
1990; Konda 2002, Section~7.6). It hides two things: the $O(\|\cdot\|^2)$ terms of the drift, and the
fact that the chain is generated under $\pi_{\theta_k}$ rather than $\pi_{\hat\theta}$, an effect of order
$\|\theta_k - \hat\theta\|$ that is part of $A_{11}$ and $A_{21}$ through the derivative of the stationary law.
We take this step as a hypothesis; everything after it is proved.

\section{The loss in expected reward}
%======================================================================

\begin{theorem}\label{thm:cov}
Let $\beta_k, \gamma_k$ satisfy Assumption~1.1 of the two-time-scale note, let $H - \frac{\bar\beta}{2}I$
and $A_{22}$ be stable, and let $\tilde\Gamma$ be the long-run covariance of the effective noise
$N_k = V_k - A_{12}A_{22}^{-1}W_k = V_k - CA_c^{-1}W_k$. Under Assumption~\ref{ass:lin},
$\beta_k^{-1}\E[(\theta_k - \hat\theta)(\theta_k - \hat\theta)^\top] \to \Sigma_{\bar\beta}$, the solution of
\[
  (H - \tfrac{\bar\beta}{2})\Sigma + \Sigma(H - \tfrac{\bar\beta}{2})^\top = \tilde\Gamma .
\]
For actor-only the same holds with $\tilde\Gamma$ the long-run covariance of $V_k$.
\end{theorem}
\begin{proof}
Lemma~\ref{lem:delta} and Corollary~2.2 (with Theorem~4.1 for the multiplicative terms) of the
two-time-scale note, and Theorem~2.1 of the single-time-scale note for actor-only.
\end{proof}

\noindent\emph{Lean:} the linear-iteration statements are \texttt{tendsto\_two\_time\_scale\_slow},
\texttt{tendsto\_mtts\_slow} and \texttt{tendsto\_secondMoment\_of\_markov\_noise}.

\begin{definition}\label{def:loss}
The \emph{loss rate} of an unbiased method ($\hat\theta = \theta^*$) is
$\ell = \lim_k\beta_k^{-1}\E[J(\theta^*) - J(\theta_k)]$. To second order,
$J(\theta^*) - J(\theta^* + x) = \frac12x^\top H_Jx + O(\|x\|^3)$, so under Assumption~\ref{ass:lin}
the loss is that of the quadratic part.
\end{definition}

\begin{proposition}\label{prop:loss}
For the linearized iteration,
$\beta_k^{-1}\E\big[\frac12x_k^\top H_Jx_k\big] \to \ell = \frac12\tr(H_J\Sigma_{\bar\beta})$. The loss is
the variance of the parameter measured in the metric of the reward: for instance, if
$\Sigma = \sigma^2 H_J^{-1}$ then $\ell = \frac{n}{2}\sigma^2$.
\end{proposition}
\begin{proof}
$\E[x^\top H_Jx] = \sum_{ij}(H_J)_{ij}\E[x_ix_j] = \tr(H_J\E[xx^\top])$, which is linear and
continuous in $\E[xx^\top]$.
\end{proof}
\noindent\emph{Lean:} \texttt{tendsto\_quadLoss}.

\begin{remark}[Why not $\operatorname{Var} J(\theta_k)$]
At a maximum the reward is flat to first order, so $J(\theta_k) - J(\theta^*) = -\frac12x_k^\top H_Jx_k + \dots$
and its variance is of order $\beta_k^2$ and depends on fourth moments. The expected loss is of order
$\beta_k$ and depends only on the covariance; it is the natural measure of how far from optimal
the policy is, and it is what we compare.
\end{remark}

\begin{proposition}[Duality]\label{prop:dual}
Let $B = H - \frac{\bar\beta}{2}I$ and let $K$ solve the adjoint equation $B^\top K + KB = H_J$. Then
$\ell = \frac12\tr(K\tilde\Gamma)$: the loss is a linear functional of the noise covariance.
\end{proposition}
\begin{proof}
$\tr(H_J\Sigma) = \tr\big((B^\top K + KB)\Sigma\big) = \tr\big(K(B\Sigma + \Sigma B^\top)\big) = \tr(K\tilde\Gamma)$.
\end{proof}
\noindent\emph{Lean:} \texttt{loss\_duality}.

\paragraph{The optimal gain.} The asymptotic covariance depends on the gain. Over matrix gains
$\beta_kG$, or with Polyak--Ruppert averaging, the least covariance is $\Sigma^* = H^{-1}\tilde\Gamma H^{-\top}$
(Konda 2002, Theorem~7.9), which does not depend on rescaling the direction. The corresponding
loss is
\[
  \ell^* = \tfrac12\tr(H_J\Sigma^*) = \tfrac12\tr(K^*\tilde\Gamma),\qquad K^* = H^{-\top}H_JH^{-1} \succeq 0 .
\]
For an unbiased method the mean direction is $(1-\gamma)\nabla J$ near $\theta^*$ (Section~1), so
$H = (1-\gamma)H_J$ and $\ell^* = \frac{1}{2(1-\gamma)^2}\tr(H_J^{-1}\tilde\Gamma)$.

\begin{proposition}[Comparison principle]\label{prop:compare}
For two methods with the same $H$, $\tilde\Gamma_A \preceq \tilde\Gamma_B$ implies $\ell^*_A \le \ell^*_B$, and
$\tilde\Gamma_A = \tilde\Gamma_B$ implies equal losses for every gain. Among unbiased methods all have
the same $H$, so the comparison is between the effective noise covariances alone.
\end{proposition}
\begin{proof}
$\ell^*_B - \ell^*_A = \frac12\tr\big(K^*(\tilde\Gamma_B - \tilde\Gamma_A)\big) \ge 0$, a trace of a product of
two positive semidefinite matrices.
\end{proof}
\noindent\emph{Lean:} \texttt{optLoss\_mono} and \texttt{optLoss\_eq\_trace}.

\section{Actor-critic versus actor-only}
%======================================================================

All expectations are under the stationary law at the given $\theta$. We assume $\E[\phi\phi^\top]$ and
$\E[\psi\psi^\top]$ are invertible.

\begin{theorem}[TD(1) critic]\label{thm:td1}
Let the critic use TD(1) with features $\phi = (\psi, \bar\phi)$, for any extra features $\bar\phi$.
Then for every $\theta$ the actor-critic mean direction equals the actor-only direction with
$\lambda = 1$, $f(\theta) = (1-\gamma)\nabla J(\theta)$, and at $\theta^*$ the effective noises have the same
long-run covariance. Hence actor-critic with a TD(1) critic has exactly the loss of actor-only
(REINFORCE), for every gain.
\end{theorem}
\begin{proof}
\emph{The critic.} With $\lambda = 1$ the trace is $z_k = \sum_j\phi(Y_j)$ over the current episode up
to $k$. Summed over an episode, the TD updates telescope:
\begin{equation}\label{eq:tel}
  \sum_kz_k\big(c_{k+1}\phi(Y_{k+1}) - \phi(Y_k)\big)^\top r = -\sum_j\phi(Y_j)\phi(Y_j)^\top r,\qquad
  \sum_kz_kg(Y_k) = \sum_j\phi(Y_j)G_j .
\end{equation}
Taking stationary expectations (equivalently, using $z_k = c_kz_{k-1} + \phi_k$ and
stationarity), $A_c = -\E[\phi\phi^\top]$ and $b = \E[\phi G]$, so $r^* = \E[\phi\phi^\top]^{-1}\E[\phi G]$ is
the least-squares regression of the returns on the features.

\emph{The direction.} Since $\psi$ is the first block of $\phi$, $\E[\psi\phi^\top]$ is the first block
row of $\E[\phi\phi^\top]$, so $-CA_c^{-1} = \E[\psi\phi^\top]\E[\phi\phi^\top]^{-1} = (I\;\,0)$. Hence
$f = Cr^* = (I\;\,0)\E[\phi G] = \E[\psi G]$, the actor-only direction, for every $\theta$; in particular
the two methods have the same root $\theta^*$ and the same $H$.

\emph{The noise.} The long-run covariance of a stationary sequence is $\lim_n\frac1n\E[S_nS_n^\top]$
for its partial sums $S_n$, and is unchanged by adding terms bounded in $L^2$. At $\theta^*$ the effective
noise is $N_k = \psi_k\phi_k^\top\hat r - C\hat r + (I\;\,0)\,d_kz_k$. By \eqref{eq:tel}, over complete
episodes,
\[
  \sum_kN_k = \sum_j\psi_j\phi_j^\top\hat r + \sum_j\psi_j(G_j - \phi_j^\top\hat r) - nC\hat r
  = \sum_j\psi_jG_j - nf(\theta^*),
\]
which is the partial sum of the actor-only noise $g(Y_k)Z_k - f$. The incomplete episodes at the two
ends contribute terms bounded in $L^2$, because episode lengths are geometric. So the long-run
covariances agree.
\end{proof}
\noindent\emph{Lean:} \texttt{ScoreChain.td1\_tdSum} (TD(1) is regression of the returns),
\texttt{ScoreChain.td1\_critic\_direction} (the direction), \texttt{ScoreChain.td1\_selector}
($-CA_c^{-1}$ selects the score block), \texttt{td1\_noise\_coboundary} (the pathwise identity
$N^{\mathrm{AC}}_k = N^{\mathrm{AO}}_k + H_{k+1} - H_k$ with $H_k = v_k(z^\psi_k - \psi_k)$, the step form of
\eqref{eq:tel}) and \texttt{tendsto\_secondMoment\_of\_coboundary} (a coboundary does not change the
asymptotic covariance).

\begin{theorem}[Compatible critic]\label{thm:compat}
Let the critic use only the compatible features, $\phi = \psi$, with TD($\lambda$) for any $\lambda$.
Then for every $\theta$ the mean direction equals the actor-only direction with trace parameter
$\lambda$, and at its root the two effective noises coincide. Hence the two methods have the same
limit and the same loss, for every gain.
\end{theorem}
\begin{proof}
Given the past, the action $a_{k+1}$ is drawn from $\pi_\theta(\cdot\mid s_{k+1})$ and the score has
zero mean over actions, so $\E[z_kc_{k+1}\psi_{k+1}^\top] = 0$ and
$\E[z_k\psi_k^\top] = \E[\psi\psi^\top] + \lambda\E[c_kz_{k-1}\psi_k^\top] = \E[\psi\psi^\top]$. Hence
$A_c = -\E[\psi\psi^\top] = -C$, $r^* = C^{-1}b$, and $f = Cr^* = b = \E[g(Y_k)z_k]$, the actor-only
direction with trace $\lambda$ (the two traces coincide). At the common root $b = 0$, so $\hat r = 0$, the
actor's noise $V_k = \psi_k\psi_k^\top\hat r - C\hat r$ vanishes, $W_k = d_kz_k = g(Y_k)z_k$, and
$-CA_c^{-1} = I$. So $N_k = g(Y_k)z_k$, the actor-only noise at the same point.
\end{proof}
\noindent\emph{Lean:} \texttt{ScoreChain.compatible\_tdSum}, \texttt{ScoreChain.compatible\_critic\_direction}
and \texttt{ScoreChain.compatible\_critic\_tdMatrix}.

\begin{remark}
For $\lambda < 1$ both methods in Theorem~\ref{thm:compat} are biased: their common limit is the
root of $\E[gz]$, not $\theta^*$. A critic helps only through features beyond the compatible ones, and
only when TD($\lambda$) with $\lambda < 1$ is used; with $\lambda = 1$ it reproduces Monte Carlo
(Theorem~\ref{thm:td1}). The state features act as a learned baseline: when they can represent
$V^\pi$, any $\lambda$ is unbiased (Theorem~\ref{thm:value}).
\end{remark}

\begin{theorem}[Critics that represent the value function]\label{thm:value}
Let the critic features be $\phi = (\psi, \bar\phi)$ where the span of the state features $\bar\phi(s)$
contains $V^{\pi_\theta}$ (for instance tabular features $\bar\phi(s) = e_s$), and let the critic use
TD($\lambda$) for any $\lambda \in [0,1]$. Then the critic's fixed point is $r^* = (w, u)$ with
$\bar\phi^\top u = V^{\pi_\theta}$ and $w = \E[\psi\psi^\top]^{-1}\E[\psi A^{\pi_\theta}]$, the least-squares fit of
the advantage $A^\pi = Q^\pi - V^\pi$ on the score. Hence the actor's mean direction is
$f(\theta) = (1-\gamma)\nabla J(\theta)$ for every $\theta$: the method is unbiased for every $\lambda$, and at
$\theta^*$ the critic converges to $\hat r = (0, u)$.
\end{theorem}
\begin{proof}
Let $\delta_k = g(Y_k) + c_{k+1}V(s_{k+1}) - V(s_k)$ be the TD error of $V = V^\pi$, so that
$\E[\delta_k\mid s_k, a_k] = Q(s_k,a_k) - V(s_k) = A(s_k,a_k)$, and $\E_{a\sim\pi}[A(s,a)] = 0$. For
$r = (w, u)$ the TD error is
$d_k = \delta_k - \psi_k^\top w + c_{k+1}\psi_{k+1}^\top w$. We check $\E[z_kd_k] = 0$, which characterizes
the fixed point when $A_c$ is invertible.
(i) $\E[z_kc_{k+1}\psi_{k+1}^\top] = 0$ because the score has mean zero over the action $a_{k+1}$, drawn
given $s_{k+1}$ and the past.
(ii) Write $z_k = \phi_k + \lambda c_kz_{k-1}$. The factor $\lambda c_kz_{k-1}$ depends only on the past and on
$s_k$, and $\E[\delta_k - \psi_k^\top w\mid s_k, \text{past}] = \E_a[A(s_k, a)] - \E_a[\psi(s_k,a)]^\top w = 0$.
(iii) So $\E[z_kd_k] = \E[\phi_k(A(s_k,a_k) - \psi_k^\top w)]$. Its state block
$\E[\bar\phi(s)\,\E_a[A - \psi^\top w]]$ vanishes, and its score block vanishes by the choice of $w$.
The actor direction is then $Cr^* = \E[\psi\psi^\top]w + \E[\psi\bar\phi^\top]u = \E[\psi A] + 0 =
\E[\psi Q] = (1-\gamma)\nabla J(\theta)$, using $\E[\psi\,h(s)] = 0$ for functions of the state. At $\theta^*$,
$\E[\psi A] = 0$ gives $w = 0$.
\end{proof}
\noindent\emph{Lean:} \texttt{ScoreChain.value\_critic\_fixed\_point} and
\texttt{ScoreChain.value\_critic\_direction}.

\begin{theorem}[TD(0) with a tabular critic beats REINFORCE]\label{thm:td0}
Let the critic use TD(0) with features $\phi = (\psi, e_s)$, and let the step law of the chain depend
on $Y_k$ only through $(s_k, a_k)$ (true for the restart chain). At $\theta^*$, with
$B_k = c_kZ_{k-1}$ the part of the REINFORCE trace carried over from the past, $b(s) =
\E[B_k\mid s_k = s]$ and $\delta_k$ the TD error of $V^\pi$,
\[
  \tilde\Gamma_{\text{AO}} - \tilde\Gamma_{\text{AC}}
  = \E\big[\delta_k^2\,(B_k - b(s_k))(B_k - b(s_k))^\top\big]
  = \sum_s\nu(s)\,\E[\delta_k^2\mid s_k = s]\,\mathrm{Cov}(B_k\mid s_k = s) \succeq 0 .
\]
Hence $\ell^*_{\text{AC}} \le \ell^*_{\text{AO}}$ by Proposition~\ref{prop:compare}.
\end{theorem}
\begin{proof}
Write $\mathcal G_k$ for the history up to $Y_{k-1}$ together with $(f_k, s_k)$, before the action $a_k$
is drawn. Given $\mathcal G_k$, the law of $(a_k, Y_{k+1})$ depends only on $s_k$, while $B_k$ is
$\mathcal G_k$-measurable.

\emph{The gain.} By Theorem~\ref{thm:value}, $\hat r = (0, V)$. The TD(0) matrix is
$A_c = \begin{psmallmatrix}-\Psi & X\\ 0 & T\end{psmallmatrix}$ with
$X = \E[\psi_k(c_{k+1}e_{s_{k+1}} - e_{s_k})^\top]$ and $T = \E[e_{s_k}(c_{k+1}e_{s_{k+1}} - e_{s_k})^\top]$
(the lower-left block vanishes because the score has mean zero over $a_{k+1}$ and over $a_k$). So
$-CA_c^{-1} = (I\;\, -K)$ with $K = XT^{-1}$. Let $M = \E[B_ke_{s_k}^\top]$, whose columns are
$\nu(s)b(s)$, and $P_c(s,s') = \E[c_{k+1}1\{s_{k+1} = s'\}\mid s_k = s]$. From
$B_{k+1} = c_{k+1}(\psi_k + B_k)$ and the conditional independence, $M = X + MP_c$, while
$T = D_\nu(P_c - I)$. Hence $K = M(I - P_c)(P_c - I)^{-1}D_\nu^{-1}$, that is $Ke_s = -b(s)$.

\emph{The two noises.} The actor-critic noise at $\theta^*$ is
$N^{\text{AC}}_k = V(s_k)\psi_k + \delta_k(\psi_k - Ke_{s_k}) = V(s_k)\psi_k + \delta_k(\psi_k + b(s_k))$.
The actor-only noise $g(Y_k)Z_k$ equals $V(s_k)\psi_k + \delta_kZ_k$ up to a coboundary, as in
Theorem~\ref{thm:td1}. Since $Z_k = \psi_k + B_k$,
\[
  N^{\text{AO}}_k \simeq N^{\text{AC}}_k + D_k,\qquad D_k = \delta_k(B_k - b(s_k)).
\]

\emph{Orthogonality.} $\E[D_k\mid\mathcal G_k] = (B_k - b(s_k))\,\E[\delta_k\mid\mathcal G_k] = 0$, because
$\E[\delta_k\mid\mathcal G_k] = \E_a[A(s_k, a)] = 0$; so $D$ is a martingale difference sequence for
$(\mathcal G_k)$. Take $N^{\text{AC}}_j$ and $D_k$. If $k > j$, then $N^{\text{AC}}_j$ is $\mathcal G_k$-measurable, so
$\E[N^{\text{AC}}_jD_k^\top] = 0$. If $k \le j$, then $\E[N^{\text{AC}}_j\mid\mathcal G_{k+1}]\,\delta_k$ has a
conditional mean given $(\mathcal G_k, a_k)$ that is a function $h(s_k, a_k)$. This holds because
$\E[N^{\text{AC}}_j\mid\mathcal G_j]$ depends only on $s_j$ when $j > k$. So
$\E[N^{\text{AC}}_jD_k^\top] = \E[h(s_k, a_k)(B_k - b(s_k))^\top] = 0$, since given $s_k$ the action $a_k$ is
independent of $B_k$, and $B_k - b(s_k)$ has conditional mean zero. Every cross-covariance therefore
vanishes, and the long-run covariances add:
$\tilde\Gamma_{\text{AO}} = \tilde\Gamma_{\text{AC}} + \E[D_kD_k^\top]$. The factored form of $\E[D_kD_k^\top]$
uses the same conditional independence, since $\delta_k^2$ depends only on $(s_k, a_k, Y_{k+1})$.
\end{proof}
\noindent\emph{Lean:} \texttt{ScoreChain.td0\_gain} ($Ke_s = -b(s)$, in the form $\sum_sT_{ss'}b(s) = -X_{s'}$),
\texttt{ScoreChain.sum\_carry\_centered} (the orthogonality: $\E[w(s_k,a_k)(B_k - b(s_k))] = 0$) and
\texttt{td0\_noise\_split} (the pathwise splitting
$g z = N^{\text{AC}} + \delta(Z - \psi - b) - (H' - H)$). The martingale step is
\texttt{StateDriven.tendsto\_secondMoment\_of\_centered\_split}. Its hypotheses are:
\begin{itemize}
\item the filtration $(\mathcal G_k)$, under which the law of the next observation depends on the past
  only through the state;
\item $\E[\delta\mid\mathcal G_k] = 0$;
\item $\E[(B_k - b(s))1\{s_k = s\}] \to 0$.
\end{itemize}
Under these, $D$ is a martingale difference orthogonal to every function of $(s_k, Y_{k+1})$, and the
limit covariances add. Two things are not formalized: that the restart chain with the filtration
$(\mathcal G_k)$ satisfies these hypotheses, and the factored form of $\E[DD^\top]$.

\begin{lemma}[The gain of the tabular TD($\lambda$) critic]\label{lem:gain}
Under the hypotheses of Theorem~\ref{thm:td0}, for every $\lambda \in [0,1]$ the state block of
$-CA_c^{-1} = (I\;\, -K_\lambda)$ is $K_\lambda e_s = -(1-\lambda)\,b(s)$.
\end{lemma}
\begin{proof}
For $\lambda = 0$ this is the gain in Theorem~\ref{thm:td0}; for $\lambda = 1$ it is Theorem~\ref{thm:td1}.
Let $0 < \lambda < 1$ and write $z_k = (z^\psi_k, z^e_k)$ for the critic's trace. Let
$R_k = z^\psi_k - \psi_k = \lambda c_kz^\psi_{k-1}$, and define
\[
P_\lambda = \E[R_ke_{s_k}^\top],\qquad E_\lambda = \E[z^e_ke_{s_k}^\top],\qquad M = \E[B_ke_{s_k}^\top].
\]
The columns of $M$ are $\nu(s)b(s)$; let $D_\nu$ be the diagonal matrix of the $\nu(s)$.
\begin{itemize}
\item \emph{The blocks of $A_c$.} As in Theorem~\ref{thm:td0}, $A_c =
  \begin{psmallmatrix}-\Psi & X_\lambda\\ 0 & T_\lambda\end{psmallmatrix}$ and $K_\lambda = X_\lambda T_\lambda^{-1}$, where
  $X_\lambda = \E[z^\psi_k(c_{k+1}e_{s_{k+1}} - e_{s_k})^\top]$ and $T_\lambda = \E[z^e_k(c_{k+1}e_{s_{k+1}} - e_{s_k})^\top]$.
  From $z^e_{k+1} = \lambda c_{k+1}z^e_k + e_{s_{k+1}}$, $R_{k+1} = \lambda c_{k+1}z^\psi_k$ and stationarity,
  \[
    T_\lambda = \lambda^{-1}(E_\lambda - D_\nu) - E_\lambda,\qquad X_\lambda = (\lambda^{-1} - 1)P_\lambda,
  \]
  using $\E[\psi_ke_{s_k}^\top] = 0$.
\item \emph{A candidate gain.} Define
  $\tilde X_k = R_k + (1-\lambda)\sum_{j\le k}\lambda^{k-j}\Pi_{jk}\,b(s_j)$, where $\Pi_{jk} = \prod_{i=j+1}^kc_i$ is
  the indicator that $j$ and $k$ lie in the same episode. The proof of Theorem~\ref{thm:mono} shows
  $\E[\tilde X_k\mid s_k] = b(s_k)$, that is, $\E[\tilde X_ke_{s_k}^\top] = M$. Since
  $\sum_{j\le k}\lambda^{k-j}\Pi_{jk}b(s_j) = MD_\nu^{-1}z^e_k$, this reads
  $P_\lambda + (1-\lambda)MD_\nu^{-1}E_\lambda = M$.
\item \emph{It solves the critic's equation.} For $\tilde K = -(1-\lambda)MD_\nu^{-1}$,
  \begin{align*}
    \lambda\tilde KT_\lambda &= -(1-\lambda)MD_\nu^{-1}\big((1-\lambda)E_\lambda - D_\nu\big)\\
    &= (1-\lambda)\big(M - (1-\lambda)MD_\nu^{-1}E_\lambda\big) = (1-\lambda)P_\lambda = \lambda X_\lambda .
  \end{align*}
  So $\tilde K = K_\lambda$, because $T_\lambda$ is invertible.
\end{itemize}
\end{proof}

\begin{theorem}[Shorter traces help the value-function critic]\label{thm:mono}
Under the hypotheses of Theorem~\ref{thm:td0}, let $\tilde\Gamma(\lambda)$ be the noise covariance of
actor-critic with the tabular TD($\lambda$) critic at $\theta^*$. Then $\tilde\Gamma(\lambda)$ is nondecreasing in
$\lambda$ in the Loewner order. More precisely, there are matrices $\Delta_i \succeq 0$, not depending on
$\lambda$, with
\[
  \tilde\Gamma(\lambda) = \tilde\Gamma(0) + \sum_{i\ge1}\lambda^{2i}\Delta_i ,
\]
where
$\Delta_i = \sum_s\nu(s)\,\E[\delta_k^2\mid s_k = s]\;\E[(Y_i - Y_{i-1})(Y_i - Y_{i-1})^\top\mid s_k = s]$
and the $Y_i$ are defined in the proof. Hence the optimal-gain loss $\ell^*$ is nondecreasing in
$\lambda$. TD(0) is the best tabular critic, and TD(1), which equals REINFORCE, is the worst.
\end{theorem}
\begin{proof}
By Theorem~\ref{thm:value}, $\hat r = (0, V)$ for every $\lambda$.

\emph{The noise.} The effective noise is
$N_k(\lambda) = V(s_k)\psi_k + \delta_k(z^\psi_k - K_\lambda z^e_k)$. By Lemma~\ref{lem:gain} and
$z^e_k = \sum_{j\le k}\lambda^{k-j}\Pi_{jk}e_{s_j}$,
\[
N_k(\lambda) = V(s_k)\psi_k + \delta_k(\psi_k + X_{\lambda,k}),\qquad
X_{\lambda,k} = R_k + (1-\lambda)\sum_{j\le k}\lambda^{k-j}\Pi_{jk}b(s_j).
\]
In particular $X_{0,k} = b(s_k)$.

\emph{A martingale in the lookback.} For $l \ge 0$ let $\mathcal F_l$ be generated by
$s_{k-l}, a_{k-l}, Y_{k-l+1},\dots,Y_{k-1}$ and $(f_k, s_k)$: the last $l$ steps before $k$, without the flag
at time $k-l$. These $\sigma$-algebras increase with $l$. Split
$B_k = \sum_{j=k-l}^{k-1}\Pi_{jk}\psi_j + \Pi_{k-l,k}B_{k-l}$. Given $s_{k-l}$, the variables generating
$\mathcal F_l$ are independent of $B_{k-l}$, because the action is drawn afresh and the step law does
not depend on the flag. Hence
\[
Y_l := \E[B_k\mid\mathcal F_l] = \sum_{j=k-l}^{k-1}\Pi_{jk}\psi_j + \Pi_{k-l,k}\,b(s_{k-l}),
\qquad Y_0 = b(s_k),\qquad Y_l \to B_k .
\]
\emph{The trace is a geometric average.} Summing the geometric weights $(1-\lambda)\lambda^l$ gives
$X_{\lambda,k} = \sum_{l\ge0}(1-\lambda)\lambda^lY_l = Y_0 + \sum_{i\ge1}\lambda^i(Y_i - Y_{i-1})$. The increments
$Y_i - Y_{i-1}$ are martingale differences for $(\mathcal F_l)$, and $s_k \in \mathcal F_0$. So
$\E[X_{\lambda,k}\mid s_k] = b(s_k)$, and
\[
\E\big[(X_{\lambda,k} - b(s_k))(X_{\lambda,k} - b(s_k))^\top\mid s_k\big]
= \sum_{i\ge1}\lambda^{2i}\,\E[(Y_i - Y_{i-1})(Y_i - Y_{i-1})^\top\mid s_k].
\]
\emph{The covariances add.} $N_k(\lambda) = N_k(0) + \delta_k(X_{\lambda,k} - b(s_k))$. The second term has the
form of $D$ in Theorem~\ref{thm:td0}, with $X_{\lambda,k}$ in place of $B_k$: it is known before the action
$a_k$ and has conditional mean $b(s_k)$ given $s_k$. The argument of Theorem~\ref{thm:td0} therefore gives
$\tilde\Gamma(\lambda) = \tilde\Gamma(0) + \E[\delta_k^2(X_{\lambda,k} - b(s_k))(X_{\lambda,k} - b(s_k))^\top]$. The conditional
independence factors this expectation into the stated sum. Monotonicity follows term by term, and
the statement about $\ell^*$ follows by Proposition~\ref{prop:compare}.
\end{proof}
\noindent\emph{Lean:} the step from the splitting to the sum of covariances is
\texttt{StateDriven.tendsto\_secondMoment\_of\_centered\_split}, with $B_k = X_{\lambda,k}$. Lemma~\ref{lem:gain}
and the lookback representation are not formalized. Numerical checks:
\begin{itemize}
\item \texttt{experiments/check\_lambda\_gap.py}, on 23 instances and $\lambda \in \{0, 0.2,\dots,1\}$:
  $\E[X_\lambda\mid s] = b(s)$ to $5\times10^{-14}$; the formula for
  $\tilde\Gamma(\lambda) - \tilde\Gamma(0)$ to $2\times10^{-14}$; and $\mathrm{Cov}(X_\lambda\mid s)$ increasing in $\lambda$.
\item \texttt{experiments/check\_lambda\_gain.py}: Lemma~\ref{lem:gain}, to $5\times10^{-14}$.
\item \texttt{experiments/check\_lambda\_tower.py} confirms the increment structure exactly:
  $\E[X_aX_b^\top\mid s] = \E[X_mX_m^\top\mid s]$ with $m = \sqrt{ab}$, to $10^{-14}$.
\end{itemize}

\begin{remark}[Not a projection]
For $0 < \lambda' < \lambda$ the noise $N(\lambda')$ is \emph{not} an orthogonal projection of $N(\lambda)$, because
$\E[X_{\lambda'}X_\lambda^\top\mid s] \ne \E[X_{\lambda'}X_{\lambda'}^\top\mid s]$. This is why a Monte Carlo
regression of $N^{\text{AO}}$ on $N(0.5)$ gives coefficients 1.05--1.11 rather than 1
(\texttt{experiments/check\_projection.py}). $N(0)$ is a projection of every $N(\lambda)$
(\texttt{experiments/check\_projection\_lambda.py}, exact to $10^{-14}$), which is the case $\lambda' = 0$.
In the instances computed, the loss at $\lambda = 0$ relative to REINFORCE is 0.29--0.54 in the seven
two-parameter instances of the table below, and $\mathrm{tr}\,\tilde\Gamma(0)/\mathrm{tr}\,\tilde\Gamma_{\text{AO}}$ is
between 0.10 and 0.91 (median 0.45) over 69 instances. With random extra features, which do not
represent $V^\pi$, and $\lambda < 1$, the critic is biased, with a gap $J^* - J(\hat\theta)$ of about
$2\times10^{-2}$. Whether it pays then depends on the horizon, through gap${}+\ell^*/k$.
\end{remark}

\begin{remark}[Finite time]\label{rem:finite}

\begin{itemize}
\item The covariance of the linearized iterations, Theorem~\ref{thm:cov}: fully formalized for the
  linear iterations (single and two time scales, additive and multiplicative Markov noise), with the
  noise hypotheses verified for finite-chain and trace-driven noise.
\item Propositions~\ref{prop:loss}, \ref{prop:dual} and~\ref{prop:compare}: formalized.
\item Theorems~\ref{thm:td1}, \ref{thm:compat}, \ref{thm:value}, the stationary identities of
  Theorem~\ref{thm:td0}, and the martingale step of Theorems~\ref{thm:td0} and~\ref{thm:mono}
  (\texttt{OrthogonalNoise.lean}): formalized in
  \texttt{CriticTheorems.lean} for a finite model of the sampling
  chain, \texttt{ScoreChain}: a chain on (restart flag, state, action) whose next action is drawn from
  the policy given the next state. This covers the restart chain and any other such chain. Two
  facts carry all the proofs: a function with mean zero over actions has
  $\E[c(Y_{k+1})F(Y_{k+1})\mid Y_k] = 0$ (\texttt{sum\_Q\_c\_eq\_zero}), and its correlation with the
  trace is its correlation with the current features (\texttt{sum\_mul\_trace\_eq}). The stationary
  trace moments $m(y) = \E[z_k1\{Y_k = y\}]$ enter through the recursion that characterizes them,
  $m(y') = \nu(y')\phi(y') + \lambda c(y')\sum_yQ(y,y')m(y)$. \texttt{ScoreChain.tendsto\_traceMoment}
  derives this recursion from the process. Along the chain, the moments of the trace converge
  to a solution whenever the law of $Y_k$ converges and $\sum_{y'}Q(y,y')|\lambda c(y')| < 1$.
  For the restart chain this condition holds with bound $|\lambda|\gamma$
  (\texttt{ScoreChain.restart\_contraction}). The law converges too: restarts draw from a fixed
  distribution with probability at least $1 - \gamma$, so the law satisfies
  $P_{k+1} = LP_k + b$, with $L$ an $\ell^1$ contraction by $\gamma$. Its limit is a stationary
  $\nu$ (\texttt{ScoreChain.tendsto\_law\_restart}). Hence
  \texttt{ScoreChain.tendsto\_traceMoment\_of\_restart} rests on the model alone, for every
  $|\lambda| \le 1$.
\item The noise covariance $\tilde\Gamma$ of Theorem~\ref{thm:cov}, for trace-driven noise, is the limit
  computed from the stationary first and second trace moments
  (\texttt{tendsto\_secondMoment\_of\_trace\_noise\_stationary}, in the companion note). It is no
  longer a hypothesis. This includes the unbounded REINFORCE trace. For the restart chain the
  whole statement rests on the model, in
  \texttt{ScoreChain.tendsto\_secondMoment\_of\_restart\_noise\_centered}:
  \begin{itemize}
  \item the stationary law exists;
  \item the Poisson solution $U$ exists, as the fixed point of a row contraction by $|\lambda|\gamma$
    (\texttt{exists\_poisson\_U});
  \item the Poisson solution $\hat h$ exists whenever the noise is centered under $\nu$
    (\texttt{exists\_poisson\_h}).
  \end{itemize}
  The centering condition is the stationary mean of the noise,
  $\sum_{y,y'}Q(y,y')(F(y,y')m(y) + \nu(y)e(y,y')) = 0$ (\texttt{centering\_eq\_noise\_mean}).
  At the root of the mean direction this holds by definition.
\item Assumption~\ref{ass:lin} is a hypothesis, and Lemma~\ref{lem:delta} (an implicit-function
  computation) is not formalized.
\end{itemize}
