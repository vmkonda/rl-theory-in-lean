\begin{abstract}
For the coupled linear iterations
$x_{k+1} = x_k - \beta_k(A_{11}x_k + A_{12}y_k) + \beta_kV_k$ and
$y_{k+1} = y_k - \gamma_k(A_{21}x_k + A_{22}y_k) + \gamma_kW_k$ with $\beta_k/\gamma_k \to 0$
and noise generated by a Markov chain, we prove that $\beta_k^{-1}\E[x_kx_k^\top]$ converges to
the solution of the Lyapunov equation of the single-time-scale iteration
$x_{k+1} = x_k - \beta_k\Delta x_k + \beta_k(V_k - A_{12}A_{22}^{-1}W_k)$, with
$\Delta = A_{11} - A_{12}A_{22}^{-1}A_{21}$. This extends Theorems~7.6 and~7.8 of
Konda~(2002), proved there for i.i.d.\ noise, to Markov noise. As in the single-time-scale
note, the proof reduces Markov noise to martingale differences with the Poisson equation and
then studies deterministic recursions for the second moments; the new ingredient is a lemma
on three coupled recursions running on two time scales. This is step~2 of the asymptotic
covariance analysis of actor-critic algorithms.
\end{abstract}

\section{Setting and assumptions}
%======================================================================

Let $(\Omega, \F, (\F_k)_{k\ge0}, \mathbb P)$ be a filtered probability space and consider,
for $x_k \in \R^{d_1}$ and $y_k \in \R^{d_2}$,
\begin{equation}\label{eq:iter}
\begin{aligned}
  x_{k+1} &= x_k - \beta_k(A_{11}x_k + A_{12}y_k) + \beta_kV_k,\\
  y_{k+1} &= y_k - \gamma_k(A_{21}x_k + A_{22}y_k) + \gamma_kW_k,
\end{aligned}
\end{equation}
with $(x_0, y_0)$ $\F_0$-measurable and square integrable and $(V_k, W_k)$
$\F_{k+1}$-measurable. When $A_{22}$ is invertible, let
\[
  D = A_{22}^{-1}A_{21},\qquad \Delta = A_{11} - A_{12}A_{22}^{-1}A_{21} = A_{11} - A_{12}D .
\]
The fast iterate $y_k$ tracks $-Dx_k$, the solution of $A_{21}x + A_{22}y = 0$, and with
$y = -Dx$ the slow iteration becomes $x_{k+1} = x_k - \beta_k\Delta x_k + \beta_kV_k$.
Norms of matrices are operator norms.

\begin{assumption}[Step sizes]\label{ass:step}
$\beta_k, \gamma_k > 0$ are deterministic and tend to zero;
$\beta_{k+1}^{-1} - \beta_k^{-1} \to \bar\beta \ge 0$ and $\gamma_{k+1}^{-1} - \gamma_k^{-1} \to 0$;
and
\[
  \frac{\beta_k}{\gamma_k} \to 0,\qquad \frac{\gamma_k^2}{\beta_k} \to 0 .
\]
\end{assumption}

For example $\beta_k = \alpha/(k + k_0)$ and $\gamma_k = c/(k + k_0)^\rho$ with
$\frac12 < \rho < 1$, which gives $\bar\beta = 1/\alpha$.

\begin{assumption}[Stability]\label{ass:stable}
The eigenvalues of $A_{22}$ and of $\Delta_{\bar\beta} := \Delta - \frac{\bar\beta}{2}I$ have positive real
parts.
\end{assumption}

\begin{assumption}[Noise]\label{ass:noise}
The noise decomposes as
\[
  V_k = M_{k+1} + h_k - h_{k+1},\qquad W_k = M'_{k+1} + g_k - g_{k+1},
\]
where $h_k, g_k$ are $\F_k$-measurable, $M_{k+1}, M'_{k+1}$ are $\F_{k+1}$-measurable with
$\E[M_{k+1}\mid\F_k] = \E[M'_{k+1}\mid\F_k] = 0$, all four have bounded second moments, and
\[
  \E[M_{k+1}M_{k+1}^\top] \to \Gamma_{11},\qquad
  \E[M_{k+1}M'^\top_{k+1}] \to \Gamma_{12} = \Gamma_{21}^\top,\qquad
  \E[M'_{k+1}M'^\top_{k+1}] \to \Gamma_{22}.
\]
\end{assumption}

By Section~4 of the single-time-scale note, Assumption~\ref{ass:noise} holds when $(V_k, W_k)$
is a function of a finite Markov chain or of a chain with eligibility traces, and then
$\Gamma = \begin{psmallmatrix}\Gamma_{11} & \Gamma_{12}\\ \Gamma_{21} & \Gamma_{22}\end{psmallmatrix}$ is the
long-run covariance of $(V_k, W_k)$.

\begin{remark}[The condition $\gamma_k^2 = o(\beta_k)$]\label{rem:gamma2}
With martingale-difference noise ($h = g = 0$) this condition is not needed. With Markov
noise the telescoping part of $W_k$ shifts the fast iterate by $\gamma_kg_k$, and we bound its
correlation with the slow iterate, of size $\sqrt{\beta_k}$, by Cauchy--Schwarz; the
condition makes that bound $o(\beta_k)$. It holds for the usual schedules above.
\end{remark}

\section{Main result}
%======================================================================

Let $\hat y_k = y_k + Dx_k$ be the deviation of the fast iterate from its quasi-static value.

\begin{theorem}\label{thm:main}
Under Assumptions~\ref{ass:step}--\ref{ass:noise},
\[
  \frac{\E[x_kx_k^\top]}{\beta_k} \to \Sigma_{11},\qquad
  \frac{\E[x_k\hat y_k^\top]}{\beta_k} \to \Sigma_{12},\qquad
  \frac{\E[\hat y_k\hat y_k^\top]}{\gamma_k} \to \Sigma_{22},
\]
where $\Sigma_{22}$, $\Sigma_{12}$, $\Sigma_{11}$ are the unique solutions of
\begin{align}
  A_{22}\Sigma_{22} + \Sigma_{22}A_{22}^\top &= \Gamma_{22}, \label{eq:s22}\\
  \Sigma_{12}A_{22}^\top + A_{12}\Sigma_{22} &= \Gamma_{12}, \label{eq:s12}\\
  \Delta_{\bar\beta}\Sigma_{11} + \Sigma_{11}\Delta_{\bar\beta}^\top + A_{12}\Sigma_{12}^\top + \Sigma_{12}A_{12}^\top
    &= \Gamma_{11}. \label{eq:s11}
\end{align}
\end{theorem}

\noindent\emph{Lean:} \texttt{tendsto\_two\_time\_scale\_covariance}, fully formalized. There $D$ is
any matrix with $A_{22}D = A_{21}$, stability takes the positive-definite form of Remark~1.3 of the
single-time-scale note for $A_{22}$ and $\Delta_{\bar\beta}$, and the noise hypotheses are as in
Theorem~2.1 of that note.

These are equations (7.14)--(7.16) of Konda (2002), there for i.i.d.\ noise. Eliminating
$\Sigma_{12}$ and $\Sigma_{22}$ gives the analogue of Theorem~7.8 there.

\begin{corollary}\label{cor:eff}
$\Sigma_{11}$ is the unique solution of
\[
  \Delta_{\bar\beta}\Sigma_{11} + \Sigma_{11}\Delta_{\bar\beta}^\top = \tilde\Gamma,\qquad
  \tilde\Gamma = \Gamma_{11} - A_{12}A_{22}^{-1}\Gamma_{21} - \Gamma_{12}A_{22}^{-\top}A_{12}^\top
    + A_{12}A_{22}^{-1}\Gamma_{22}A_{22}^{-\top}A_{12}^\top .
\]
$\tilde\Gamma$ is the long-run covariance of the effective noise $V_k - A_{12}A_{22}^{-1}W_k$, so
the slow iterate has the asymptotic covariance of the single-time-scale iteration
$x_{k+1} = x_k - \beta_k\Delta x_k + \beta_k(V_k - A_{12}A_{22}^{-1}W_k)$ (Theorem~2.1 of the
single-time-scale note).
\end{corollary}

\begin{remark}[Numerical check]
For a 6-state chain with $d_1 = d_2 = 1$, $A = \begin{psmallmatrix}1 & 0.8\\ -0.5 & 1.5\end{psmallmatrix}$,
$\beta_k = \alpha/(k + 50)$ with $\alpha = 2/\Delta$ and $\gamma_k = (k + 50)^{-0.7}$, a simulation with
$20{,}000$ runs gives $\beta_k^{-1}\E[x_k^2] \approx 1.131$ at $k = 10^5$, against
$\Sigma_{11} = 1.142$; the noise $V$ alone would give $0.606$
(\texttt{experiments/check\_two\_time\_scale.py}).
\end{remark}

\section{Proof}
%======================================================================

Throughout, $C$ denotes a constant independent of $k$, and $\epsilon_k$ a sequence tending to
zero; both may change from line to line. By Assumption~\ref{ass:step},
$\beta_k\le\gamma_k$ for large $k$; we assume this for all $k$, which changes nothing.

\subsection{Change of variables}

\begin{lemma}\label{lem:cov}
Let $\tilde x_k = x_k + \beta_kh_k$ and $\tilde y_k = \hat y_k + \gamma_kg_k + \beta_kDh_k$. Then
\begin{align*}
  \tilde x_{k+1} &= \tilde x_k - \beta_k(\Delta\tilde x_k + A_{12}\tilde y_k) + \beta_kM_{k+1} + c_k,\\
  \tilde y_{k+1} &= \tilde y_k - \gamma_kA_{22}\tilde y_k - \beta_kD(\Delta\tilde x_k + A_{12}\tilde y_k)
    + \gamma_kM'_{k+1} + \beta_kDM_{k+1} + c'_k,
\end{align*}
with $\E\|c_k\|^2 \le C\beta_k^2\gamma_k^2$ and $\E\|c'_k\|^2 \le C\gamma_k^4$.
\end{lemma}
\begin{proof}
Since $A_{21}x + A_{22}y = A_{22}\hat y$ and $A_{11}x + A_{12}y = \Delta x + A_{12}\hat y$,
\begin{align*}
  x_{k+1} &= x_k - \beta_k(\Delta x_k + A_{12}\hat y_k) + \beta_kV_k,\\
  \hat y_{k+1} &= \hat y_k - \gamma_kA_{22}\hat y_k - \beta_kD(\Delta x_k + A_{12}\hat y_k) + \gamma_kW_k + \beta_kDV_k .
\end{align*}
Now substitute the decompositions of $V_k$ and $W_k$ and proceed as in Lemma~2.3 of the
single-time-scale note. One finds
\begin{align*}
  c_k &= \beta_k\big(\beta_k\Delta h_k + \gamma_kA_{12}g_k + \beta_kA_{12}Dh_k\big) + (\beta_{k+1} - \beta_k)h_{k+1},\\
  c'_k &= \gamma_kA_{22}(\gamma_kg_k + \beta_kDh_k) + \beta_kD\big(\beta_k\Delta h_k + \gamma_kA_{12}g_k + \beta_kA_{12}Dh_k\big)\\
    &\quad + (\gamma_{k+1} - \gamma_k)g_{k+1} + (\beta_{k+1} - \beta_k)Dh_{k+1},
\end{align*}
and the bounds follow from $|\beta_{k+1} - \beta_k| \le C\beta_k^2$, $|\gamma_{k+1} - \gamma_k| \le C\gamma_k^2$
and $\beta_k \le \gamma_k$.
\end{proof}
\noindent\emph{Lean:} \texttt{ttsX\_succ}, \texttt{ttsY\_succ} (the identities),
\texttt{norm\_tts\_cx\_le}, \texttt{norm\_tts\_cy\_le} (pointwise bounds on $c_k$, $c'_k$).

\subsection{Second-moment recursions}

Let $X_k = \E[\tilde x_k\tilde x_k^\top]$, $Z_k = \E[\tilde x_k\tilde y_k^\top]$ and
$Y_k = \E[\tilde y_k\tilde y_k^\top]$, and
\[
  n_k = 1 + \frac{\|X_k\|}{\beta_k} + \frac{\|Z_k\|}{\beta_k} + \frac{\|Y_k\|}{\gamma_k}.
\]

\begin{lemma}\label{lem:rec}
\begin{align*}
  X_{k+1} &= X_k - \beta_k\big(\Delta X_k + X_k\Delta^\top + A_{12}Z_k^\top + Z_kA_{12}^\top\big)
    + \beta_k^2\Gamma_{11} + R_k, & \|R_k\| &\le \epsilon_k\beta_k^2n_k,\\
  Z_{k+1} &= Z_k - \gamma_kZ_kA_{22}^\top - \beta_kA_{12}Y_k + \beta_k\gamma_k\Gamma_{12} + R'_k,
    & \|R'_k\| &\le \epsilon_k\beta_k\gamma_kn_k,\\
  Y_{k+1} &= Y_k - \gamma_k\big(A_{22}Y_k + Y_kA_{22}^\top\big) + \gamma_k^2\Gamma_{22} + R''_k,
    & \|R''_k\| &\le \epsilon_k\gamma_k^2n_k .
\end{align*}
\end{lemma}
\begin{proof}
Write $\tilde x_{k+1} = a + b + c$ and $\tilde y_{k+1} = a' + b' + c'$, with $a, a'$ the
$\F_k$-measurable parts, $b = \beta_kM_{k+1}$ and $b' = \gamma_kM'_{k+1} + \beta_kDM_{k+1}$
the martingale differences, and $c = c_k$, $c' = c'_k$. For any two of these sums,
\[
  \E[(a + b + c)(a' + b' + c')^\top] = \E[aa'^\top] + \E[bb'^\top] + \rho,
\]
where $\E[ab'^\top] = \E[ba'^\top] = 0$ by the tower property and $\rho$ collects the five terms
involving $c$ or $c'$, each bounded by Cauchy--Schwarz. We use
$\E\|\tilde x_k\|^2 \le d_1\|X_k\| \le C\beta_kn_k$ and $\E\|\tilde y_k\|^2 \le C\gamma_kn_k$.

\emph{The $Z$ recursion.} With $w = \Delta\tilde x_k + A_{12}\tilde y_k$,
$a = \tilde x_k - \beta_kw$ and $a' = \tilde y_k - \gamma_kA_{22}\tilde y_k - \beta_kDw$, so
\[
  \E[aa'^\top] = Z_k - \gamma_kZ_kA_{22}^\top - \beta_k\E[w\tilde y_k^\top] - \beta_k\E[\tilde x_kw^\top]D^\top
    + \beta_k\gamma_k\E[ww'^\top] + \beta_k^2\E[ww^\top]D^\top
\]
with $w' = A_{22}\tilde y_k$. Here $\E[w\tilde y_k^\top] = \Delta Z_k + A_{12}Y_k$; the term
$\beta_k\Delta Z_k$ is at most $\beta_k^2n_k = (\beta_k/\gamma_k)\beta_k\gamma_kn_k$, and
$\|\E[\tilde x_kw^\top]\| \le C\beta_kn_k$ in the same way. Further
$\E\|w\|^2 \le C\gamma_kn_k$, so $\beta_k\gamma_k\|\E[ww'^\top]\| \le C\beta_k\gamma_k^2n_k$ and
$\beta_k^2\|\E[ww^\top]\| \le C\beta_k^2\gamma_kn_k$. The martingale part is
$\E[bb'^\top] = \beta_k\gamma_k\E[M_{k+1}M'^\top_{k+1}] + \beta_k^2\E[M_{k+1}M_{k+1}^\top]D^\top$, and
$\E[M_{k+1}M'^\top_{k+1}] - \Gamma_{12} \to 0$. Finally, by Lemma~\ref{lem:cov},
\[
  \|\rho\| \le C\Big(\sqrt{\beta_kn_k}\,\gamma_k^2 + \beta_k\gamma_k^2 + \beta_k\gamma_k\sqrt{\gamma_kn_k}
    + \beta_k\gamma_k^2 + \beta_k\gamma_k^3\Big)
  \le C\beta_k\gamma_k\Big(\frac{\gamma_k}{\sqrt{\beta_k}} + \sqrt{\gamma_k}\Big)n_k,
\]
using $\sqrt{n} \le n$; this is where $\gamma_k^2/\beta_k \to 0$ is used.

\emph{The $X$ and $Y$ recursions} are similar and simpler. For $X$, the term
$\beta_k^2\E[ww^\top]$ is at most $C\beta_k^2\gamma_kn_k$, and the cross terms with $c$ are at most
$C\sqrt{\beta_kn_k}\,\beta_k\gamma_k \le C\beta_k^2(\gamma_k/\sqrt{\beta_k})n_k$. For $Y$, the terms
of order $\beta_k$ are at most $C\beta_k\gamma_kn_k = (\beta_k/\gamma_k)\gamma_k^2n_k$, and the cross
terms with $c'$ are at most $C\sqrt{\gamma_kn_k}\,\gamma_k^2$.
\end{proof}
\noindent\emph{Lean:} \texttt{tts\_remainder\_step}, from the one-step expansions
\texttt{tts\_step\_X}, \texttt{tts\_step\_Z}, \texttt{tts\_step\_Y} (built on
\texttt{crossMoment\_step}), the size bounds \texttt{tts\_sizes}, and the arithmetic of
\texttt{tts\_R1\_le}, \texttt{tts\_R2\_le}, \texttt{tts\_R3\_le}.

\subsection{A deterministic lemma for coupled recursions}

\begin{lemma}\label{lem:coupled}
Let $\Lop_1, \Lop_2, \Lop_3$ be linear maps on finite-dimensional real inner product spaces
$\mathcal V_1, \mathcal V_2, \mathcal V_3$ with $\langle\Lop_iv, v\rangle \ge \eta\|v\|^2$ for some
$\eta > 0$, and let $K_{12}\colon \mathcal V_2\to\mathcal V_1$ and $K_{23}\colon\mathcal V_3\to\mathcal V_2$
be linear. Let $\beta_k, \gamma_k \to 0$ be positive with $\sum_k\beta_k = \infty$ and
$\beta_k \le \gamma_k$. Suppose
\begin{align*}
  a_{k+1} &= a_k - \beta_k(\Lop_1a_k + K_{12}c_k - g_1) + \beta_k\delta^1_k,\\
  c_{k+1} &= c_k - \gamma_k(\Lop_2c_k + K_{23}b_k - g_2) + \gamma_k\delta^2_k,\\
  b_{k+1} &= b_k - \gamma_k(\Lop_3b_k - g_3) + \gamma_k\delta^3_k,
\end{align*}
with $\|\delta^i_k\| \le \epsilon_k(1 + \|a_k\| + \|b_k\| + \|c_k\|)$ and $\epsilon_k \to 0$. Then
$b_k \to b^*$, $c_k \to c^*$ and $a_k \to a^*$, the unique solutions of
$\Lop_3b^* = g_3$, $\Lop_2c^* + K_{23}b^* = g_2$ and $\Lop_1a^* + K_{12}c^* = g_1$.
\end{lemma}
\begin{proof}
The solutions exist and are unique because each $\Lop_i$ is coercive, hence invertible. Let
$e^a = a - a^*$, $e^c = c - c^*$, $e^b = b - b^*$. Then
\begin{align*}
  e^a_{k+1} &= (I - \beta_k\Lop_1)e^a_k + \beta_k(\delta^1_k - K_{12}e^c_k),\\
  e^c_{k+1} &= (I - \gamma_k\Lop_2)e^c_k + \gamma_k(\delta^2_k - K_{23}e^b_k),\\
  e^b_{k+1} &= (I - \gamma_k\Lop_3)e^b_k + \gamma_k\delta^3_k ,
\end{align*}
and $\|\delta^i_k\| \le \epsilon_k(S + \|e^a_k\| + \|e^b_k\| + \|e^c_k\|)$ with
$S = 1 + \|a^*\| + \|b^*\| + \|c^*\|$. For small $t$, $\|(I - t\Lop_i)v\| \le (1 - \eta t/2)\|v\|$.

\emph{Boundedness.} Let $\kappa = \eta/2$, $K_b = 1/\kappa$,
$K_c = (\|K_{23}\|K_b + 1)/\kappa$, $K_a = (\|K_{12}\|K_c + 1)/\kappa$ and
$s = S + K_a + K_b + K_c$. Take $k_0$ such that for $k \ge k_0$ the contraction bounds hold
and $\epsilon_ks \le 1$, and take $\lambda \ge 1$ with $\|e^a_{k_0}\| \le \lambda K_a$,
$\|e^b_{k_0}\| \le \lambda K_b$, $\|e^c_{k_0}\| \le \lambda K_c$. If these three bounds hold at
$k \ge k_0$, then $\|\delta^i_k\| \le \epsilon_k\lambda s \le \lambda$, and
\begin{align*}
  \|e^b_{k+1}\| &\le (1 - \kappa\gamma_k)\lambda K_b + \gamma_k\lambda = \lambda K_b,\\
  \|e^c_{k+1}\| &\le (1 - \kappa\gamma_k)\lambda K_c + \gamma_k(\lambda + \|K_{23}\|\lambda K_b) = \lambda K_c,
\end{align*}
and similarly $\|e^a_{k+1}\| \le \lambda K_a$. So $a_k, b_k, c_k$ are bounded and $\delta^i_k \to 0$.

\emph{Convergence.} Lemma~2.7 of the single-time-scale note (with step sizes $\gamma_k$,
$\sum\gamma_k \ge \sum\beta_k = \infty$) gives $e^b_k \to 0$; then the perturbation
$\delta^2_k - K_{23}e^b_k \to 0$ gives $e^c_k \to 0$; then $e^a_k \to 0$ in the same way.
\end{proof}
\noindent\emph{Lean:} \texttt{tendsto\_zero\_of\_coupled\_recursion}; the normalization of the next
subsection is \texttt{tendsto\_two\_time\_scale\_moments}.

\subsection{Proof of Theorem~\ref{thm:main}}

Let $a_k = X_k/\beta_k$, $c_k = Z_k/\beta_k$, $b_k = Y_k/\gamma_k$, so $n_k = 1 + \|a_k\| + \|b_k\| + \|c_k\|$.
Write $\beta_k/\beta_{k+1} = 1 + \beta_k(\bar\beta + \epsilon_k)$ and
$\gamma_k/\gamma_{k+1} = 1 + \gamma_k\epsilon'_k$ with $\epsilon_k, \epsilon'_k \to 0$. Dividing the
recursions of Lemma~\ref{lem:rec} by $\beta_{k+1}$, $\beta_{k+1}$ and $\gamma_{k+1}$:
\begin{align*}
  a_{k+1} &= a_k - \beta_k\big(\Delta_{\bar\beta}a_k + a_k\Delta_{\bar\beta}^\top + A_{12}c_k^\top + c_kA_{12}^\top - \Gamma_{11}\big)
    + \beta_k\delta^1_k,\\
  c_{k+1} &= c_k - \gamma_k\big(c_kA_{22}^\top + A_{12}b_k - \Gamma_{12}\big) + \gamma_k\delta^2_k,\\
  b_{k+1} &= b_k - \gamma_k\big(A_{22}b_k + b_kA_{22}^\top - \Gamma_{22}\big) + \gamma_k\delta^3_k ,
\end{align*}
where, for instance,
$\delta^2_k = \frac{\beta_k}{\beta_{k+1}}\frac{R'_k}{\beta_k\gamma_k} + \frac{\beta_k}{\gamma_k}(\bar\beta + \epsilon_k)
\big(c_k - \gamma_k(c_kA_{22}^\top + A_{12}b_k - \Gamma_{12})\big)$. The factors $\beta_k/\gamma_k$,
$\gamma_k$, $\epsilon_k$ and $\epsilon'_k$ all tend to zero, so
$\|\delta^i_k\| \le \epsilon_k n_k$. The three maps
$\Lop_1a = \Delta_{\bar\beta}a + a\Delta_{\bar\beta}^\top$, $\Lop_2c = cA_{22}^\top$ and
$\Lop_3b = A_{22}b + bA_{22}^\top$ are coercive in the inner products of
Remark~1.3 of the single-time-scale note, so Lemma~\ref{lem:coupled} applies with
$K_{12}c = A_{12}c^\top + cA_{12}^\top$ and $K_{23}b = A_{12}b$, and gives the three limits for
the shifted iterates. Finally $x_k = \tilde x_k - \beta_kh_k$ and
$\hat y_k = \tilde y_k - \gamma_kg_k - \beta_kDh_k$, and as at the end of the proof of Theorem~2.1
of the single-time-scale note the differences are of lower order.
\qed

\subsection{Proof of Corollary~\ref{cor:eff}}

By \eqref{eq:s12}, $\Sigma_{12} = (\Gamma_{12} - A_{12}\Sigma_{22})A_{22}^{-\top}$, so
\[
  A_{12}\Sigma_{12}^\top + \Sigma_{12}A_{12}^\top
  = A_{12}A_{22}^{-1}\Gamma_{21} + \Gamma_{12}A_{22}^{-\top}A_{12}^\top
    - A_{12}\big(A_{22}^{-1}\Sigma_{22} + \Sigma_{22}A_{22}^{-\top}\big)A_{12}^\top,
\]
and by \eqref{eq:s22}, $A_{22}^{-1}\Sigma_{22} + \Sigma_{22}A_{22}^{-\top}
= A_{22}^{-1}(\Sigma_{22}A_{22}^\top + A_{22}\Sigma_{22})A_{22}^{-\top} = A_{22}^{-1}\Gamma_{22}A_{22}^{-\top}$.
Substituting in \eqref{eq:s11} gives the equation for $\Sigma_{11}$. The effective noise
$V_k - A_{12}A_{22}^{-1}W_k$ has the Poisson decomposition with martingale part
$M_{k+1} - A_{12}A_{22}^{-1}M'_{k+1}$, whose covariance tends to $\tilde\Gamma$.
\qed

\section{What remains}
%======================================================================

\begin{itemize}
\item \emph{Multiplicative noise on two time scales.} In actor-critic both $A_{12}$ (the critic
  enters the actor through the features) and $A_{22}$ (the TD matrix) are random. Section~3 of
  the single-time-scale note shows how to absorb such terms; the same shift applies here.
\item \emph{Nonlinear algorithms.} The transfer to the actual actor-critic iteration is left as
  an explicit hypothesis.
\end{itemize}
