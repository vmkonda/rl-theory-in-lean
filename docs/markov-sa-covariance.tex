\title{Asymptotic Covariance of Linear Stochastic Approximation\\ with Markov Noise}
\author{Vijaymohan Konda}
\date{November 2026}

\begin{document}
\maketitle

\begin{abstract}
For the linear iteration $\theta_{k+1} = \theta_k - \beta_kA\theta_k + \beta_kV_k$ with
noise $V_k$ generated by a Markov chain, we prove that
$\beta_k^{-1}\E[\theta_k\theta_k^\top]$ converges to the solution $\Sigma$ of a Lyapunov
equation whose right-hand side is the long-run covariance $\Gamma$ of the noise. The
proof reduces Markov noise to martingale-difference noise with the Poisson equation,
and then analyses a deterministic recursion for the second moments, as in the
i.i.d.\ case. Only two lemmas involve probability; the rest is matrix analysis. We
verify the hypotheses for noise driven by a finite chain and for noise driven by
eligibility traces. Finally, we show that multiplicative Markov noise, a random matrix
$A_k$ in place of $A$ as in TD($\lambda$), does not change the limit. These are the first and
third of four steps towards the asymptotic covariance of actor-critic algorithms: single time
scale with additive noise, two time scales, multiplicative noise, and the transfer to the
nonlinear algorithm.
\end{abstract}

\section{Setting and assumptions}
%======================================================================

Let $(\Omega, \F, (\F_k)_{k\ge0}, \mathbb P)$ be a filtered probability space,
$A \in \R^{d\times d}$, and consider
\begin{equation}\label{eq:iter}
  \theta_{k+1} = \theta_k - \beta_kA\theta_k + \beta_kV_k, \qquad k \ge 0,
\end{equation}
with $\theta_0$ $\F_0$-measurable and square integrable, and $V_k$
$\F_{k+1}$-measurable. Norms of matrices are operator norms.

\begin{assumption}[Step sizes]\label{ass:step}
$\beta_k > 0$ is deterministic, $\beta_k \to 0$, and
$\beta_{k+1}^{-1} - \beta_k^{-1} \to \bar\beta$ for some $\bar\beta \in [0,\infty)$.
\end{assumption}

For example $\beta_k = \alpha/(k + k_0)$ gives $\bar\beta = 1/\alpha$, and
$\beta_k = \alpha/(k+k_0)^\rho$ with $\rho < 1$ gives $\bar\beta = 0$. In both cases
$\sum_k\beta_k = \infty$; this holds in general under Assumption~\ref{ass:step}, since
$\beta_k^{-1} \le \beta_0^{-1} + k(\bar\beta + 1)$ for large $k$.

\begin{assumption}[Stability]\label{ass:stable}
Every eigenvalue of $A_{\bar\beta} := A - \tfrac{\bar\beta}{2}I$ has positive real part.
\end{assumption}

\begin{remark}[Form used in Lean]\label{rem:posdef}
The Lean formalization assumes instead that $A_{\bar\beta}$ is positive definite in the
asymmetric sense, $x^\top A_{\bar\beta}x \ge \eta\|x\|^2$ for some $\eta > 0$
(\texttt{Matrix.PosDefAsymm}). This is stronger in a fixed basis but loses no generality:
with $B$ as in Lemma~\ref{lem:B}, the coordinates $\tilde\theta = B^{1/2}\theta$ turn
Assumption~\ref{ass:stable} into this form, and the covariance transforms as
$\tilde\Sigma = B^{1/2}\Sigma B^{1/2}$. In exchange, the Lyapunov operator is coercive in the
ordinary Frobenius inner product (\texttt{lyapOp\_coercive}), which avoids spectral theory.
\end{remark}

\begin{assumption}[Noise]\label{ass:noise}
The noise decomposes as
\[
  V_k = M_{k+1} + h_k - h_{k+1},
\]
where $h_k$ is $\F_k$-measurable with $\sup_k\E\|h_k\|^2 < \infty$, $M_{k+1}$ is
$\F_{k+1}$-measurable with $\E[M_{k+1}\mid\F_k] = 0$ and $\sup_k\E\|M_{k+1}\|^2 < \infty$,
and
\[
  \Gamma_k := \E[M_{k+1}M_{k+1}^\top] \longrightarrow \Gamma .
\]
\end{assumption}

Section~\ref{sec:verify} shows that Assumption~\ref{ass:noise} holds for noise
generated by a finite Markov chain and by eligibility traces, with $\Gamma$ the
long-run covariance $\sum_{l\in\mathbb Z}\E[V_0V_l^\top]$.

\section{Main result}
%======================================================================

\begin{theorem}\label{thm:main}
Under Assumptions~\ref{ass:step}--\ref{ass:noise},
\[
  \frac{1}{\beta_k}\,\E[\theta_k\theta_k^\top] \longrightarrow \Sigma,
\]
where $\Sigma$ is the unique solution of the Lyapunov equation
\begin{equation}\label{eq:lyap}
  A_{\bar\beta}\Sigma + \Sigma A_{\bar\beta}^\top = \Gamma .
\end{equation}
\end{theorem}

\noindent\emph{Lean:} \texttt{tendsto\_secondMoment\_of\_markov\_noise}, fully formalized.
There, stability takes the form of Remark~\ref{rem:posdef}; the martingale property is stated
componentwise, $\E[(M_{k+1})_j\mid\F_k] = 0$; and the moment conditions are
$h_k, M_{k+1} \in L^2$ with $\sup_k\E\|h_k\|^2 < \infty$ and $\sup_k\E\|M_{k+1}\|^2 < \infty$.

The proof occupies the rest of this section. Throughout, $C$ denotes a constant that
does not depend on $k$ and may change from line to line.

\subsection{Step sizes}

\begin{lemma}\label{lem:step}
Under Assumption~\ref{ass:step}, write $\beta_k/\beta_{k+1} = 1 + \beta_k(\bar\beta + \epsilon_k)$.
Then $\epsilon_k \to 0$, $\beta_{k+1}/\beta_k \to 1$, and $|\beta_{k+1} - \beta_k| \le C\beta_k^2$.
\end{lemma}
\begin{proof}
The identity $\beta_k/\beta_{k+1} = 1 + \beta_k(\beta_{k+1}^{-1} - \beta_k^{-1})$ shows
$\epsilon_k = (\beta_{k+1}^{-1} - \beta_k^{-1}) - \bar\beta \to 0$, and then
$\beta_k/\beta_{k+1}\to1$ because $\beta_k \to 0$. Finally
$|\beta_k - \beta_{k+1}| = \beta_k\beta_{k+1}|\beta_{k+1}^{-1} - \beta_k^{-1}|$, where the
last factor is bounded and $\beta_{k+1} \le C\beta_k$.
\end{proof}
\noindent\emph{Lean:} \texttt{StepSizeRate}, \texttt{StepSizeRate.tendsto\_ratio},
\texttt{StepSizeRate.abs\_sub\_le}, and \texttt{StepSizeRate.tendsto\_sum} for
$\sum_k\beta_k = \infty$.

\subsection{Removing the telescoping part of the noise}

\begin{lemma}\label{lem:cov}
Let $\theta'_k = \theta_k + \beta_kh_k$. Then
\[
  \theta'_{k+1} = (I - \beta_kA)\theta'_k + \beta_kM_{k+1} + \varepsilon_{k+1},
  \qquad
  \varepsilon_{k+1} = \beta_k^2Ah_k + (\beta_{k+1} - \beta_k)h_{k+1},
\]
and $\E\|\varepsilon_{k+1}\|^2 \le C\beta_k^4$.
\end{lemma}
\begin{proof}
Substituting the decomposition of $V_k$ in \eqref{eq:iter},
\[
  \theta'_{k+1} = \theta_{k+1} + \beta_{k+1}h_{k+1}
  = \theta_k - \beta_kA\theta_k + \beta_kM_{k+1} + \beta_kh_k + (\beta_{k+1} - \beta_k)h_{k+1}.
\]
Now $\theta_k + \beta_kh_k = \theta'_k$ and
$-\beta_kA\theta_k = -\beta_kA\theta'_k + \beta_k^2Ah_k$. The bound follows from
Lemma~\ref{lem:step} and $\sup_k\E\|h_k\|^2 < \infty$.
\end{proof}
\noindent\emph{Lean:} \texttt{shifted\_iterate} (the identity, pointwise); adaptedness and
square integrability of the iterates are \texttt{iterates\_adapted}.

\subsection{The second-moment recursion}

\begin{lemma}\label{lem:rec}
Let $P_k = \E[\theta'_k\theta'^\top_k]$. Then
\[
  P_{k+1} = (I - \beta_kA)P_k(I - \beta_kA)^\top + \beta_k^2\Gamma_k + R_k,
  \qquad
  \|R_k\| \le C\beta_k^2\big(\sqrt{\tr P_k} + \beta_k\big).
\]
\end{lemma}
\begin{proof}
Write $\theta'_{k+1} = a + b + c$ with $a = (I - \beta_kA)\theta'_k$, $b = \beta_kM_{k+1}$,
$c = \varepsilon_{k+1}$. Since $a$ is $\F_k$-measurable and $\E[M_{k+1}\mid\F_k] = 0$,
the tower property gives $\E[ab^\top] = 0$. Hence
$P_{k+1} = \E[aa^\top] + \E[bb^\top] + R_k$ with
$R_k = \E[ac^\top + ca^\top + bc^\top + cb^\top + cc^\top]$, and $\E[bb^\top] = \beta_k^2\Gamma_k$.
By Cauchy--Schwarz, $\|\E[ac^\top]\| \le \sqrt{\E\|a\|^2\,\E\|c\|^2} \le C\sqrt{\tr P_k}\,\beta_k^2$,
$\|\E[bc^\top]\| \le C\beta_k\cdot\beta_k^2$ and $\|\E[cc^\top]\| \le C\beta_k^4$, using
Lemma~\ref{lem:cov}.
\end{proof}
\noindent\emph{Lean:} \texttt{secondMoment\_recursion}, built on \texttt{crossMoment\_eq\_zero}
(tower property, via the pull-out property of conditional expectation),
\texttt{norm\_crossMoment\_le} (Cauchy--Schwarz), \texttt{secondMoment\_sub\_smul} (the term
$\E[aa^\top]$), and \texttt{integral\_norm\_sq\_le} ($\E\|x\|^2 \le d\,\|\E[xx^\top]\|$, used in
place of $\tr P_k$).

\subsection{An a priori bound}

\begin{lemma}[Lyapunov matrix]\label{lem:B}
There is $B \succ 0$ with $A_{\bar\beta}^\top B + BA_{\bar\beta} = I$, i.e.\
$A^\top B + BA = I + \bar\beta B$.
\end{lemma}
\begin{proof}
$-A_{\bar\beta}$ is Hurwitz, so $B = \int_0^\infty e^{-A_{\bar\beta}^\top t}e^{-A_{\bar\beta}t}\,dt$
converges; it is positive definite, and differentiating
$e^{-A_{\bar\beta}^\top t}e^{-A_{\bar\beta}t}$ and integrating gives the equation.
\end{proof}

\begin{lemma}\label{lem:apriori}
$\sup_k \tr(P_k)/\beta_k < \infty$.
\end{lemma}
\begin{proof}
Let $x_k = \tr(BP_k)$, so that $\lambda_{\min}(B)\tr P_k \le x_k \le \lambda_{\max}(B)\tr P_k$.
Using Lemma~\ref{lem:B},
\begin{align*}
  \tr\!\big(B(I - \beta A)P(I - \beta A)^\top\big)
  &= \tr\!\big(P\,[B - \beta(A^\top B + BA) + \beta^2A^\top BA]\big) \\
  &= x - \beta\tr P - \beta\bar\beta x + \beta^2\tr(PA^\top BA).
\end{align*}
With $a = 1/\lambda_{\max}(B) > 0$ we have $\tr P \ge ax$, so by Lemma~\ref{lem:rec}
\[
  x_{k+1} \le \big(1 - \beta_k(\bar\beta + a) + C\beta_k^2\big)x_k + C\beta_k^2\big(1 + \sqrt{x_k}\big).
\]
Let $y_k = x_k/\beta_k$ and use $\beta_k/\beta_{k+1} = 1 + \beta_k(\bar\beta + \epsilon_k)$
(Lemma~\ref{lem:step}) and $\beta_k\sqrt{x_k} = \beta_k\sqrt{\beta_ky_k} \le \beta_k(1 + \beta_ky_k)$:
\begin{align*}
  y_{k+1} &\le \big(1 + \beta_k(\bar\beta + \epsilon_k)\big)\big(1 - \beta_k(\bar\beta + a) + C\beta_k^2\big)y_k
    + C\beta_k \\
  &= \big(1 - \beta_k(a - \epsilon_k + O(\beta_k))\big)y_k + C\beta_k .
\end{align*}
Choose $k_1$ such that for $k \ge k_1$ the coefficient is at most $1 - a\beta_k/2$ and
$a\beta_k/2 \le 1$. Then $Y = \max(y_{k_1}, 2C/a)$ satisfies: $y_k \le Y$ implies
$y_{k+1} \le (1 - a\beta_k/2)Y + C\beta_k \le Y$. By induction $y_k \le Y$ for $k \ge k_1$.
\end{proof}
\noindent\emph{Lean:} \texttt{norm\_le\_of\_second\_moment\_recursion}, with the scalar
induction in \texttt{bdd\_of\_le\_contraction}. The Lean proof works with the Frobenius norm
$\|P_k\|$ instead of $\tr(BP_k)$, with the remainder bound stated as
$\|R_k\| \le C\beta_k^2(\sqrt{\|P_k\|} + \|P_k\| + \beta_k)$ (the middle term, which is needed
for multiplicative noise in Section~\ref{sec:mult}, is absorbed into the $\beta_k^2AP_kA^\top$
term); it then needs the sharper one-step bound
$\|S - \beta\mathcal AS\| \le (1 - a\beta + \tfrac12\beta^2\|\mathcal A\|^2)\|S\|$
(\texttt{norm\_sub\_smul\_le'}) where $a = 2\eta + \bar\beta$ is the coercivity constant of
$\mathcal A$, but no positive semidefiniteness of $P_k$.

\subsection{A deterministic lemma}

\begin{lemma}\label{lem:pert}
Let $\Lop$ be a linear map on a finite-dimensional real vector space whose
eigenvalues all have positive real part. Let $\beta_k \to 0$ with $\sum_k\beta_k = \infty$,
let $\delta_k \to 0$, and let $e_{k+1} = (I - \beta_k\Lop)e_k + \beta_k\delta_k$. Then $e_k \to 0$.
\end{lemma}
\begin{proof}
As in Lemma~\ref{lem:B} there is an inner product $\langle\cdot,\cdot\rangle_*$ with
$\langle\Lop v, v\rangle_* \ge a\|v\|_*^2$ for some $a > 0$. Then
$\|(I - \beta\Lop)v\|_*^2 = \|v\|_*^2 - 2\beta\langle\Lop v, v\rangle_* + \beta^2\|\Lop v\|_*^2
\le (1 - 2a\beta + C\beta^2)\|v\|_*^2$, so $\|I - \beta_k\Lop\|_* \le 1 - a\beta_k$ for large $k$, and
\[
  \|e_{k+1}\|_* \le (1 - a\beta_k)\|e_k\|_* + \beta_k\|\delta_k\|_* .
\]
Fix $\eta > 0$ and $k_2$ with $\|\delta_k\|_* \le a\eta$ for $k \ge k_2$. Then
$u_k = \|e_k\|_* - \eta$ satisfies $u_{k+1} \le (1 - a\beta_k)u_k$ for $k\ge k_2$. If
$u_k \le 0$ for some $k \ge k_2$ it stays so; otherwise
$u_k \le u_{k_2}\prod_{j=k_2}^{k-1}(1 - a\beta_j) \to 0$ because $\sum\beta_j = \infty$.
Hence $\limsup\|e_k\|_* \le \eta$ for every $\eta > 0$.
\end{proof}
\noindent\emph{Lean:} \texttt{tendsto\_zero\_of\_coercive\_recursion} (for a coercive
continuous linear map on a real inner product space), built on
\texttt{norm\_sub\_smul\_le} and the scalar lemma \texttt{tendsto\_zero\_of\_le\_contraction};
coercivity of $S \mapsto AS + SA^\top$ is \texttt{lyapOp\_coercive}.

\subsection{Proof of Theorem~\ref{thm:main}}

Let $S_k = P_k/\beta_k$ and $\mathcal A S = AS + SA^\top$. By Lemma~\ref{lem:rec},
\[
  \frac{P_{k+1}}{\beta_k} = S_k - \beta_k\mathcal AS_k + \beta_k^2AS_kA^\top
    + \beta_k\Gamma_k + \beta_k\frac{R_k}{\beta_k^2}.
\]
Multiplying by $\beta_k/\beta_{k+1} = 1 + \beta_k(\bar\beta + \epsilon_k)$,
\[
  S_{k+1} = S_k - \beta_k\big(\mathcal AS_k - \bar\beta S_k - \Gamma\big) + \beta_k\delta_k,
\]
where
\[
  \delta_k = \epsilon_kS_k + (\Gamma_k - \Gamma) + \beta_kAS_kA^\top + \frac{R_k}{\beta_k^2}
    + \beta_k(\bar\beta + \epsilon_k)\Big(-\mathcal AS_k + \beta_kAS_kA^\top + \Gamma_k + \frac{R_k}{\beta_k^2}\Big).
\]
By Lemma~\ref{lem:apriori}, $S_k$ is bounded and $\tr P_k = O(\beta_k)$, so
$\|R_k\|/\beta_k^2 \le C(\sqrt{\tr P_k} + \beta_k) \to 0$; with $\epsilon_k \to 0$ and
$\Gamma_k \to \Gamma$, every term of $\delta_k$ tends to zero.

Define $\Lop S = \mathcal AS - \bar\beta S = A_{\bar\beta}S + SA_{\bar\beta}^\top$ on symmetric
matrices. Its eigenvalues are the sums $\lambda_i + \lambda_j$ of eigenvalues of
$A_{\bar\beta}$, which have positive real part; in particular $\Lop$ is invertible and
\eqref{eq:lyap} has a unique solution $\Sigma$. With $e_k = S_k - \Sigma$,
\[
  e_{k+1} = (I - \beta_k\Lop)e_k + \beta_k\delta_k ,
\]
so $e_k \to 0$ by Lemma~\ref{lem:pert} (recall $\sum\beta_k = \infty$).

Finally, $\theta_k = \theta'_k - \beta_kh_k$, so
\[
  \big\|\E[\theta_k\theta_k^\top] - P_k\big\| \le 2\beta_k\sqrt{\E\|h_k\|^2\,\tr P_k} + \beta_k^2\E\|h_k\|^2
  = O(\beta_k^{3/2}),
\]
and therefore $\beta_k^{-1}\E[\theta_k\theta_k^\top] - S_k \to 0$. \qed

\smallskip\noindent\emph{Lean:} the part of this proof that starts from the second-moment
recursion of Lemma~\ref{lem:rec} is \texttt{tendsto\_normalized\_second\_moment}, using the
exact formula for $\delta_k$ in terms of $P_k$ and \texttt{tendsto\_zero\_of\_coercive\_recursion};
the solution $\Sigma$ of \eqref{eq:lyap} exists by \texttt{exists\_solution\_of\_coercive}.
The remaining, probabilistic, steps are Lemmas~\ref{lem:cov} and~\ref{lem:rec}.

\section{Multiplicative noise}\label{sec:mult}
%======================================================================

In TD($\lambda$) and actor-critic methods the matrix is itself random. Consider
\begin{equation}\label{eq:mult}
  \theta_{k+1} = \theta_k - \beta_k(A + \tilde A_k)\theta_k + \beta_kV_k,
\end{equation}
with $V_k$ as in Assumption~\ref{ass:noise} and a random matrix $\tilde A_k$ of mean zero
in the following sense.

\begin{assumption}[Multiplicative noise]\label{ass:mult}
The matrix noise decomposes as
\[
  \tilde A_k = N_{k+1} + G_k - G_{k+1},
\]
where $G_k$ is $\F_k$-measurable, $N_{k+1}$ is $\F_{k+1}$-measurable with
$\E[N_{k+1}\mid\F_k] = 0$ entrywise, and $\|N_{k+1}\|, \|G_k\| \le C$ almost surely.
\end{assumption}

For a finite chain, $\tilde A_k = \tilde a(Y_k, Y_{k+1})$ with $\E_\nu\tilde a = 0$ has this
decomposition by Lemma~\ref{lem:poisson} applied to each entry, and $N$ and $G$ are bounded
because the state space is finite. With traces, $A_k = z_k(\phi(Y_k) - \gamma\phi(Y_{k+1}))^\top$
is affine in $z_k$, and Lemma~\ref{lem:trace-poisson} gives the decomposition; boundedness holds
when the traces are bounded, for instance when $\sup_yc(y) < 1$.

\begin{theorem}\label{thm:mult}
Under Assumptions~\ref{ass:step}, \ref{ass:stable} (for $A$), \ref{ass:noise}
and~\ref{ass:mult}, the iterates of \eqref{eq:mult} satisfy
$\beta_k^{-1}\E[\theta_k\theta_k^\top] \to \Sigma$, where $\Sigma$ solves \eqref{eq:lyap} with
the same $\Gamma = \lim\E[M_{k+1}M_{k+1}^\top]$ as in Theorem~\ref{thm:main}.
\end{theorem}

\noindent\emph{Lean:} \texttt{tendsto\_secondMoment\_of\_mult\_noise}, fully formalized, with
the hypotheses in the form described after Theorem~\ref{thm:main}, $N_{k+1}$ and $G_k$ taken as
random continuous linear maps with $\sup\|N_{k+1}\|, \sup\|G_k\| < \infty$ pointwise, and the
martingale property of $N_{k+1}$ stated as $\E[(N_{k+1}e_i)_j\mid\F_k] = 0$ for all $i, j$.

\begin{remark}[Interpretation]
In TD($\lambda$) with $b_k$ the reward part, $\theta$ is measured from the limit $\theta^*$, so
$V_k = b_k - A_k\theta^*$ is the noise at the fixed point. The theorem says that the
fluctuation of $A_k$ around its mean acts on $\theta_k - \theta^* = O(\sqrt{\beta_k})$ and is
therefore of lower order: only the noise at $\theta^*$ enters the covariance.
\end{remark}

The proof follows Section~2. The new point is that the telescoping part $G_k - G_{k+1}$ of the
matrix noise is removed by a shift that depends on $\theta_k$, and the martingale part
$N_{k+1}\theta_k$ is a martingale difference whose variance vanishes with $\theta_k$.

\begin{lemma}\label{lem:mult-shift}
Let $u_k = \theta_k + \beta_k(h_k - G_k\theta_k)$. Then
\[
  u_{k+1} = (I - \beta_kA)u_k + \beta_k\big(M_{k+1} - N_{k+1}\theta_k\big) + c_k,
\]
\[
  c_k = \beta_k^2A(h_k - G_k\theta_k) + (\beta_{k+1} - \beta_k)(h_{k+1} - G_{k+1}\theta_{k+1})
    - \beta_kG_{k+1}(\theta_{k+1} - \theta_k),
\]
and $\|c_k\| \le C\beta_k^2\big(\|\theta_k\| + \|M_{k+1}\| + \|h_k\| + \|h_{k+1}\|\big)$.
If moreover $\beta_kC \le \tfrac12$, then $\|\theta_k\| \le 2\|u_k\| + 2\beta_k\|h_k\|$.
\end{lemma}
\begin{proof}
Write $\tilde A_k\theta_k = N_{k+1}\theta_k + G_k\theta_k - G_{k+1}\theta_{k+1}
+ G_{k+1}(\theta_{k+1} - \theta_k)$ and proceed as in Lemma~\ref{lem:cov}: the terms
$G_k\theta_k$ and $G_{k+1}\theta_{k+1}$ telescope into the shift, and
$-\beta_kA\theta_k = -\beta_kAu_k + \beta_k^2A(h_k - G_k\theta_k)$. For the bound, by
\eqref{eq:mult}, $\|\theta_{k+1} - \theta_k\| \le C\beta_k(\|\theta_k\| + \|V_k\|)$, so also
$\|\theta_{k+1}\| \le C(\|\theta_k\| + \|V_k\|)$; together with $|\beta_{k+1} - \beta_k| \le C\beta_k^2$
this bounds each term of $c_k$. Finally
$\|\theta_k\| \le \|u_k\| + \beta_k\|h_k\| + \beta_kC\|\theta_k\|$.
\end{proof}
\noindent\emph{Lean:} \texttt{mult\_shifted\_iterate} (the identity), \texttt{norm\_mult\_remainder\_le}
and \texttt{norm\_le\_of\_multShift} (the bounds).

\begin{lemma}\label{lem:mult-rec}
Assume $\beta_kC \le \tfrac12$ for all $k$, and let $P_k = \E[u_ku_k^\top]$. Then
\[
  P_{k+1} = (I - \beta_kA)P_k(I - \beta_kA)^\top + \beta_k^2\Gamma_k + R_k,
  \qquad
  \|R_k\| \le C\beta_k^2\big(\sqrt{\|P_k\|} + \|P_k\| + \beta_k\big).
\]
\end{lemma}
\begin{proof}
$N_{k+1}\theta_k$ is a martingale difference: its $j$-th component is
$\sum_i(\theta_k)_i(N_{k+1})_{ji}$, and $\E[(\theta_k)_i(N_{k+1})_{ji}\mid\F_k]
= (\theta_k)_i\E[(N_{k+1})_{ji}\mid\F_k] = 0$. So $m_{k+1} = M_{k+1} - N_{k+1}\theta_k$ is a
martingale difference, and as in Lemma~\ref{lem:rec},
\[
  P_{k+1} = (I - \beta_kA)P_k(I - \beta_kA)^\top + \beta_k^2\E[m_{k+1}m_{k+1}^\top] + R'_k,
\]
\[
  \|R'_k\| \le 2\sqrt{\E\|(I - \beta_kA)u_k\|^2\,\E\|c_k\|^2} + 2\beta_k\sqrt{\E\|m_{k+1}\|^2\,\E\|c_k\|^2}
    + \E\|c_k\|^2 .
\]
By Lemma~\ref{lem:mult-shift}, $T_k := \E\|\theta_k\|^2 \le 8\E\|u_k\|^2 + 8\beta_k^2\E\|h_k\|^2
\le C(\|P_k\| + \beta_k^2)$, and
\[
  \E\|c_k\|^2 \le C\beta_k^4(T_k + 1),\qquad
  \E\|m_{k+1}\|^2 \le C(1 + T_k),\qquad
  \big\|\E[m_{k+1}m_{k+1}^\top] - \Gamma_k\big\| \le C\big(\sqrt{T_k} + T_k\big).
\]
Since $\sqrt{T_k} \le C(\sqrt{\|P_k\|} + \beta_k)$, every term is $\beta_k^2$ times a polynomial
in $\sqrt{\|P_k\|}$, $\beta_k$ and $\sqrt{T_k}$ of degree at most two in
$(\sqrt{\|P_k\|}, \sqrt{T_k})$ and without constant term, hence
$O\big(\beta_k^2(\sqrt{\|P_k\|} + \|P_k\| + \beta_k)\big)$.
\end{proof}
\noindent\emph{Lean:} \texttt{mult\_secondMoment\_recursion}; the martingale property of
$N_{k+1}\theta_k$ is \texttt{condExp\_clm\_apply\_eq\_zero}, the expansion for a general
decomposition $u' = (I - \beta A)u + \beta m + c$ is \texttt{secondMoment\_step}, and the final
polynomial estimate is \texttt{mult\_poly\_le}.

\begin{proof}[Proof of Theorem~\ref{thm:mult}]
Since $\beta_k \to 0$, the condition $\beta_kC \le \frac12$ holds from some $k_0$ on, and
the theorem is unchanged by starting the iteration at $k_0$. Lemmas~\ref{lem:apriori}
and~\ref{lem:pert}, and the proof of Theorem~\ref{thm:main}, use the remainder only through
$\|R_k\|/\beta_k^2 \to 0$ once $\|P_k\| = O(\beta_k)$, and the extra term
$C\beta_k^2\|P_k\|$ is absorbed into $\beta_k^2AP_kA^\top$ in Lemma~\ref{lem:apriori}. Hence
$P_k/\beta_k \to \Sigma$. Finally $\theta_k = u_k - \beta_k(h_k - G_k\theta_k)$ and
$\E\|h_k - G_k\theta_k\|^2 \le 2\E\|h_k\|^2 + 2C^2T_k$ is bounded, so
$\beta_k^{-1}\|\E[\theta_k\theta_k^\top] - P_k\| \to 0$ as at the end of Section~2.
\end{proof}
\noindent\emph{Lean:} \texttt{tendsto\_secondMoment\_of\_mult\_noise\_of\_small} (when
$\beta_kC \le \frac12$ for all $k$) and \texttt{tendsto\_secondMoment\_of\_mult\_noise}
(the time shift, via \texttt{StepSizeRate.shift} and \texttt{Filtration.shift}).

\subsection{Noise generated by a finite Markov chain}

Let $(Y_k)$ be a Markov chain on a finite set $\Y$ with irreducible aperiodic transition
matrix $Q$ and stationary law $\nu$, $\F_k = \sigma(\theta_0, Y_0,\dots,Y_k)$, and
$V_k = v(Y_k, Y_{k+1})$ with $\sum_{y,y'}\nu(y)Q(y,y')v(y,y') = 0$. Let
$\bar v(y) = \sum_{y'}Q(y,y')v(y,y')$, let $Z = (I - Q + \mathbf 1\nu^\top)^{-1}$ be the
fundamental matrix, and $\hat h = Z\bar v$ (applied componentwise).

\begin{lemma}[Poisson equation]\label{lem:poisson}
$\hat h - Q\hat h = \bar v$, and with
$M_{k+1} = v(Y_k,Y_{k+1}) - \bar v(Y_k) + \hat h(Y_{k+1}) - (Q\hat h)(Y_k)$ and
$h_k = \hat h(Y_k)$ we have $V_k = M_{k+1} + h_k - h_{k+1}$ and $\E[M_{k+1}\mid\F_k] = 0$.
\end{lemma}
\begin{proof}
$\nu^\top\bar v = 0$ by assumption, so $\mathbf 1\nu^\top\bar v = 0$ and
$(I - Q)Z\bar v = (I - Q + \mathbf 1\nu^\top)Z\bar v = \bar v$. The decomposition follows
from $\bar v = \hat h - Q\hat h$, and $\E[M_{k+1}\mid\F_k] = \bar v(Y_k) - \bar v(Y_k) +
(Q\hat h)(Y_k) - (Q\hat h)(Y_k) = 0$ by the Markov property.
\end{proof}
\noindent\emph{Lean:} \texttt{poisson\_decomposition\_eq} (the decomposition) and
\texttt{condExp\_poissonMart\_eq\_zero} (the martingale property). The chain is described by
\texttt{FiniteMarkov}: adaptedness, a row-stochastic $Q$, and the Markov property
$\E[F(Y_k, Y_{k+1})\mid\F_k] = \sum_{y'}Q(Y_k, y')F(Y_k, y')$ as a hypothesis. Any solution $\hat h$
of the Poisson equation may be used; its existence via the fundamental matrix is not formalized.

\begin{lemma}[Long-run covariance]\label{lem:lrc}
Let $m(y) = \E[M_{k+1}M_{k+1}^\top\mid Y_k = y]$ (which does not depend on $k$) and
$\Gamma = \sum_y\nu(y)m(y)$. Then $\Gamma_k \to \Gamma$ geometrically fast for every
initial law, and
\[
  \Gamma = \sum_{l\in\mathbb Z}\E_\nu[V_0V_l^\top],
\]
the series converging absolutely.
\end{lemma}
\begin{proof}
$\Gamma_k = \sum_y\mathbb P(Y_k = y)m(y)$ and $\mathbb P(Y_k = \cdot) \to \nu$ geometrically.
For the identity, start the chain from $\nu$ and let $T_n = \sum_{k<n}V_k
= \sum_{k<n}M_{k+1} + \hat h(Y_0) - \hat h(Y_n)$. The martingale differences are
orthogonal, so $\frac1n\E[T_nT_n^\top] \to \Gamma$, the boundary terms contributing
$O(n^{-1/2})$ by Cauchy--Schwarz. On the other hand
$\frac1n\E[T_nT_n^\top] = \sum_{|l|<n}(1 - |l|/n)\E_\nu[V_0V_l^\top]$, and for $l \ge 1$,
$\E_\nu[V_0V_l^\top] = \E_\nu[V_0\,(Q^{l-1}\bar v)(Y_1)^\top]$ with
$\|Q^{l-1}\bar v\|_\infty \le C\rho^{l-1}$ for some $\rho < 1$, since $\nu^\top\bar v = 0$.
So the series converges absolutely, and the Ces\`aro means converge to its sum.
\end{proof}

\begin{corollary}\label{cor:finite}
For noise generated by a finite Markov chain, Assumption~\ref{ass:noise} holds and
Theorem~\ref{thm:main} applies with $\Gamma$ the long-run covariance of the noise.
\end{corollary}
\noindent\emph{Lean:} \texttt{tendsto\_secondMoment\_of\_finite\_markov\_noise}, with
$\Gamma = \sum_y\nu(y)m(y)$ as in Lemma~\ref{lem:lrc} and the convergence of the law of $Y_k$ to
$\nu$ as a hypothesis; $\Gamma_k \to \Gamma$ is \texttt{tendsto\_crossMoment\_finite}. The
identification of $\Gamma$ with the series $\sum_l\E_\nu[V_0V_l^\top]$ and the trace-driven case of
Section~\ref{sec:verify} are not yet formalized.

\subsection{Noise generated by eligibility traces}

In TD($\lambda$) and in actor-only and actor-critic methods the noise depends on a trace,
$f_k = F(Y_k,Y_{k+1})z_k + e(Y_k,Y_{k+1})$ with $z_k = c(Y_k)z_{k-1} + \phi(Y_k)$, where the
spectral radius of $Q\operatorname{diag}(c)$ is less than one. The pair $(Y_k, z_k)$ is a
Markov chain on an infinite state space, but its Poisson equation has an explicit
affine solution. Let $U$ and $h$ be as in the closed form for $\Gamma$ (TD($\lambda$) chapter,
Theorem~4.4, and \texttt{experiments/trace\_cov.py}):
$U = \bar F + Q\operatorname{diag}(c)U$, $w = \bar e + Q[U\phi]$, $h = Zw$.

\begin{lemma}\label{lem:trace-poisson}
$H(y,z) = U(y)z + h(y)$ solves the Poisson equation of the chain $(Y_k, z_k)$:
\[
  H(y,z) - \E\big[H(Y_1, z_1)\mid Y_0 = y, z_0 = z\big] = \bar F(y)z + \bar e(y)
    = \E[f_0\mid Y_0 = y, z_0 = z].
\]
\end{lemma}
\begin{proof}
Since $z_1 = c(Y_1)z + \phi(Y_1)$, the conditional expectation equals
$\big(Q\operatorname{diag}(c)U\big)(y)\,z + \big(Q[U\phi]\big)(y) + (Qh)(y)$. Subtracting from
$H(y,z)$ and using $U - Q\operatorname{diag}(c)U = \bar F$ and $h - Qh = w - \mathbf 1\nu^\top w = w$
(because $\nu^\top w = \E f_0 = 0$) leaves $\bar F(y)z + w(y) - (Q[U\phi])(y) = \bar F(y)z + \bar e(y)$.
\end{proof}
\noindent\emph{Lean:} \texttt{condExp\_traceMart\_eq\_zero}, in the equivalent conditional form:
if $U(y) = \sum_{y'}Q(y,y')\big(F(y,y') + c(y')U(y')\big)$ and
$\hat h(y) = \sum_{y'}Q(y,y')\big(e(y,y') + U(y')\phi(y') + \hat h(y')\big)$, then
$M_{k+1} = f_k + H(Y_{k+1}, z_{k+1}) - H(Y_k, z_k)$ is a martingale difference, for traces that are
adapted and bounded. The trace recursion is $z_{k+1} = c(Y_{k+1})z_k + \phi(Y_{k+1})$ and
$F(y, y')$ is a linear map.

\begin{corollary}\label{cor:trace}
With $h_k = H(Y_k,z_k)$ and $M_{k+1} = f_k - \E[f_k\mid\F_k] + H(Y_{k+1},z_{k+1}) -
\E[H(Y_{k+1},z_{k+1})\mid\F_k]$, trace-driven noise satisfies Assumption~\ref{ass:noise}
whenever the traces have bounded second moments (for instance $\sup_yc(y) < 1$, or
traces reset at restarts), and $\Gamma = \E[M_{k+1}M_{k+1}^\top]$ under stationarity equals
the closed-form long-run covariance of the TD($\lambda$) chapter.
\end{corollary}
\begin{proof}
The decomposition $f_k = M_{k+1} + h_k - h_{k+1}$ and the martingale property follow from
Lemma~\ref{lem:trace-poisson} exactly as in Lemma~\ref{lem:poisson}. The identification of
$\Gamma$ follows as in Lemma~\ref{lem:lrc}, using that the trace forgets its initial value
geometrically.
\end{proof}
\noindent\emph{Lean:} \texttt{tendsto\_secondMoment\_of\_trace\_noise}: Theorem~\ref{thm:main} for
trace-driven noise with bounded traces, with $\Gamma = \lim\E[M_{k+1}M_{k+1}^\top]$ as a hypothesis
(its identification with the closed form is not formalized).

\begin{remark}[Numerical checks]
(i) For a random 7-state chain, $\E_\nu[MM^\top]$ computed from Lemma~\ref{lem:poisson}
equals the long-run covariance to relative error $3\times10^{-16}$; for trace-driven noise,
the corresponding quantity from Lemma~\ref{lem:trace-poisson} equals the closed form
to $7\times10^{-16}$. (ii) For a $2\times2$ iteration with Markov noise and
$\beta_k = 2/(k + 50)$, a simulation with $20{,}000$ runs gives
$\beta_k^{-1}\E[\theta_k\theta_k^\top] \approx \begin{psmallmatrix}0.482 & -0.075\\ -0.075 & 0.518\end{psmallmatrix}$
at $k = 40{,}000$, against
$\Sigma = \begin{psmallmatrix}0.479 & -0.070\\ -0.070 & 0.523\end{psmallmatrix}$ from \eqref{eq:lyap}.
\end{remark}

\section{What remains}
%======================================================================

\begin{itemize}
\item \emph{Two time scales.} The same reduction to martingale differences applies to
  each component; the second-moment analysis then follows Konda (2002, Thm.~7.6), with
  the cross covariance $\Gamma_{12}$ of the two martingale parts. The goal is the
  analogue of Thm.~7.8 for Markov noise: the slow component has the covariance of the
  single-time-scale iteration with noise $V - A_{12}A_{22}^{-1}W$.
\item \emph{Nonlinear algorithms.} The transfer from the linearized iteration to the
  actual actor-critic algorithm is left as an explicit hypothesis.
\end{itemize}
